\subsection*{بررسی شبکه‌های مولد متخاصم (\lr{GAN})}

\begin{figure}[htpb]
	\centering
	\includegraphics[width=\textwidth]{img/Gan.png}
	\caption{مقایسه ساختار، ابعاد و جزئیات معماری شبکه‌های \lr{GAN}، \lr{DCGAN} و \lr{Conditional DCGAN}}
	\label{fig:gan_comparison}
\end{figure}

همان‌طور که در تصویر \ref{fig:gan_comparison} به تصویر کشیده شده است، ایده مدل‌های مولد متخاصم از یک ساختار ساده مبتنی بر لایه‌های تمام‌متصل به مرور زمان به شبکه‌های پیچیده پیچشی و کنترل‌پذیر (شرطی) تغییر یافته است. در این بخش، جزئیات ریاضی و ساختاری هر یک از این سه معماری را بر اساس مستندات موجود در تصویر تحلیل می‌کنیم.

\subsubsection*{\lr{Original GAN} (گودفلو - ۲۰۱۴)}
ایده اولیه شبکه‌های متخاصم مولد که در بخش \lr{A} تصویر دیده می‌شود، بر پایه یک بازی با مجموع صفر (\lr{Zero-sum game}) بین دو شبکه کاملاً مجزا شکل گرفت:

\begin{itemize}
	\item \textbf{شبکه مولد $G(z)$:} این شبکه یک بردار نویز تصادفی $z$ را از یک فضای نهان با توزیع مشخص $z \sim p_z(z)$ دریافت می‌کند. ساختار داخلی این شبکه کاملاً مبتنی بر لایه‌های تمام‌متصل (\lr{Fully Connected} یا \lr{MLP}) است. وظیفه این لایه‌ها این است که بردار ورودی را گام‌به‌گام نگاشت داده و در نهایت یک نمونه خروجی مصنوعی به نام $\hat{x} = G(z)$ تولید کنند.
	\item \textbf{شبکه متمایزگر $D(x)$:} این شبکه نیز ساختاری کاملاً تمام‌متصل دارد. ورودی آن می‌تواند یک داده واقعی $x$ (از توزیع اصلی داده‌ها $p_{data}(x)$) یا یک داده جعلی $\hat{x}$ باشد. خروجی این شبکه یک عدد اسکالر در بازه $[0, 1]$ است که نشان‌دهنده احتمال واقعی بودن تصویر ورودی است.
\end{itemize}

فرآیند آموزش این مدل از طریق یک تابع هدف مشترک به صورت زیر مدیریت می‌شود که در آن متمایزگر به دنبال بیشینه‌سازی دقت خود و مولد به دنبال فریب دادن متمایزگر (کمینه‌سازی لگاریتم خطای متمایزگر روی داده‌های جعلی) است:
$$\min_{G} \max_{D} \mathbb{E}_{x \sim p_{data}(x)}[\log D(x)] + \mathbb{E}_{z \sim p_z(z)}[\log (1 - D(G(z)))]$$

\subsubsection*{\lr{DCGAN} (رادفورد - ۲۰۱۵)}
نسخه کلاسیک گودفلو به دلیل ساختار تمام‌متصل، در پردازش و تولید تصاویر با ابعاد بزرگ و دارای بافت پیچیده به شدت ناپایدار بود. رادفورد با معرفی \lr{DCGAN} (بخش \lr{B} تصویر)، لایه‌های پیچشی را جایگزین ساختار قبلی کرد و چند قانون معماری کلیدی را توسعه داد:

\begin{itemize}
	\item \textbf{معماری مولد $G(z)$:} در این مدل، لایه‌های تمام‌متصل حذف شده و جای خود را به لایه‌های کانولوشن ترانهاده (\lr{Transposed Convolution}) با گام حرکت (\lr{Stride}) دو داده‌اند. فرآیند بازسازی تصویر از یک بردار نویز $z$ شروع شده و طی لایه‌های متوالی با ابعاد فیلتر مشخص (مثلاً ابتدا ابعاد کوچک ۴×۴ با ۱۰۲۴ کانال، سپس ۸×۸ با ۵۱۲ کانال، ۱۶×۱۶ با ۲۵۶ کانال و در نهایت ۳۲×۳۲ با ۱۲۸ کانال) به ابعاد نهایی تصویر (۶۴×۶۴×۳) می‌رسد. در تمام لایه‌های میانی از تابع فعال‌سازی \lr{ReLU} و لایه نرمال‌سازی دسته‌ای (\lr{Batch Normalization}) استفاده می‌شود و لایه نهایی به تابع \lr{Tanh} مجهز است.
	\item \textbf{معماری متمایزگر $D(x)$:} این شبکه یک مدل پیچشی عمیق معکوس است. تصویر ورودی ۶۴×۶۴ با عبور از لایه‌های کانولوشن استاندارد با گام ۲، به تدریج فشرده شده و عمق کانال‌های آن افزایش می‌یابد (۶۴، ۱۲۸، ۲۵۶ و ۵۱۲). در این شبکه لایه‌های نرمال‌سازی دسته‌ای و تابع فعال‌سازی \lr{Leaky ReLU} با نرخ آلفای \lr{۰.۲} به کار گرفته شده‌اند و لایه آخر با یک تابع \lr{Sigmoid} وضعیت نهایی را مشخص می‌کند.
\end{itemize}

\subsubsection*{\lr{Conditional DCGAN} (\lr{cDCGAN})}
اگرچه \lr{DCGAN} کیفیت تصاویر را به شدت ارتقا داد، اما خروجی آن کاملاً تصادفی بود. در کاربردهای پزشکی (مانند تفکیک تومورهای خوش‌خیم و بدخیم)، نیاز به کنترل نوع تصویر تولیدی وجود دارد. این نیاز در معماری \lr{cDCGAN} (بخش \lr{C} تصویر) برطرف شد:

\begin{itemize}
	\item \textbf{تزریق شرط در مولد $G(z,y)$:} علاوه بر بردار نویز $z$، برچسب کلاس مورد نظر $y$ نیز به شبکه داده می‌شود. این برچسب ابتدا از یک لایه جایگذاری (\lr{Embedding Lookup}) عبور کرده تا به یک بردار متراکم $e_y \in \mathbb{R}^k$ تبدیل شود. سپس این دو بردار با یکدیگر ترکیب (\lr{Concatenate}) شده و بردار جدیدی را می‌سازند که هم حاوی اطلاعات ساختاری تصادفی است و هم نوع کلاس خروجی را دیکته می‌کند. ادامه فرآیند تولید مشابه \lr{DCGAN} پیش می‌رود.
	\item \textbf{تزریق شرط در متمایزگر $D(x,y)$:} برچسب کلاس $y$ به شکل یک ماسک یا کانال فضایی هم‌اندازه با تصویر ورودی بازآرایی می‌شود. این کانال شرطی، در راستای بعد کانال‌ها به تصویر ورودی متصل می‌گردد (مثلاً برای تصویر تک‌کاناله، ورودی متمایزگر ۲ کاناله خواهد شد). متمایزگر در این حالت نه تنها کیفیت تصویر، بلکه تطابق بافت آن با برچسب ورودی را هم ارزیابی می‌کند.
\end{itemize}

ریاضیات این شبکه مستقیماً به شکل شرطی تغییر یافته و به صورت زیر بازنویسی می‌شود:
$$\min_{G} \max_{D} \mathbb{E}_{x \sim p_{data}(x), y \sim p_{data}(y)}[\log D(x, y)] + \mathbb{E}_{z \sim p_z(z), y \sim p_{data}(y)}[\log (1 - D(G(z, y), y))]$$

\subsubsection*{مقایسه ساختاری و تحلیل محدودیت‌ها}
با تکیه بر جدول مشخصات فنی ارائه شده در پایین تصویر \ref{fig:gan_comparison}، می‌توان گفت هر یک از این گام‌های تکاملی برای پاسخ به یک محدودیت جدی در لایه قبلی طراحی شده‌اند:

\begin{itemize}
	\item \textbf{نوع ورودی و کنترل‌پذیری:} مدل‌های \lr{GAN} و \lr{DCGAN} صرفاً بر پایه بردار تصادفی $z$ کار می‌کنند، به این معنی که هیچ کنترلی روی خروجی وجود ندارد. مدل \lr{cDCGAN} با ترکیب لایه جایگذاری برچسب با بردار نویز، این محدودیت بزرگ را حل کرد و کنترل‌پذیری هدایت‌شده را به شبکه اضافه نمود.
	\item \textbf{معماری شبکه‌ها و لایه‌های پیچشی:} مدل اولیه به دلیل نداشتن لایه‌های کانولوشن و استفاده از ساختار تمام‌متصل (\lr{MLP})، ویژگی‌های فضایی تصاویر را از دست می‌داد که نتیجه آن تولید خروجی‌های مات و بدون جزئیات بود. \lr{DCGAN} و \lr{cDCGAN} با ورود لایه‌های کانولوشن ترانهاده و استاندارد، توانستند الگوهای محلی و بافت‌های پیچیده تصاویر را استخراج و بازسازی کنند که منجر به بهبود چشمگیر کیفیت تصاویر شد.
	\item \textbf{پایدارسازی فرآیند آموزش:} مدل‌های اولیه \lr{GAN} به شدت مستعد پدیده ناپایداری و واگرایی بودند. در مدل‌های بعدی با اعمال استانداردهایی نظیر استفاده از \lr{Batch Normalization} در لایه‌های میانی و توابع فعال‌سازی مناسب (\lr{Leaky ReLU} در متمایزگر و \lr{ReLU}/\lr{Tanh} در مولد)، رفتار گرادیان‌ها در طول شبکه مهار شد و فرآیند آموزش به ثبات بالایی رسید.
\end{itemize}

سونوگرافی پستان زنان را  به دو کلاس خوش‌خیم و بدخیم تقسیم میکند در حالی که تعداد داده های کلاس خوش‌خیم نسبت به بدخیم بسیار بیشتر است. نامتقارنی در داده های کلاس های پزشکی امری طبیعی و متداول است اما در یادگیری ماشین باعث خطای زیاد مدل در پیشبینی داده های کلاس کوپک تر میشود.

در این سوال، از مدل آماده \lr{ResNet18} برای آموزش و اعتبارسنجی داده ها استفاده کرده و برای حل مشکل کلاس های نامتقارن، به سراغ شبکه های متخاصم مولد (GAN) میرویم

\subsubsection{مدل \, \lr{ResNet18}} \label{resnet18}

مدل \lr{ResNet18} (مخفف \lr{Residual Network} با \lr{18} لایه) یکی از کاربردی‌ترین شبکه‌های عصبی کانولوشنی (\lr{CNN}) در دنیای بینایی ماشین است. وزن‌های پیش‌فرض آن (\lr{Pre-Trained}) که روی دیتابیس عظیم \lr{ImageNet} آموزش دیده‌اند، به عنوان یک استخراج‌کننده ویژگی فوق‌العاده برای تسک‌های دیگر عمل می‌کنند. این ویژگی باعث شده این مدل برای دسته‌بندی تصاویر و همچنین کاربردهای تصویربرداری و تشخیص پزشکی بسیار مناسب باشد.

در شبکه‌های عصبی کانولوشنی کلاسیک، پردازش تصویر به صورت متوالی و لایه به لایه انجام می‌شود. به این معنی که هر لایه خروجی لایه قبلی را دریافت کرده، تغییراتی روی آن اعمال می‌کند و به لایه بعدی منتقل می‌کند. اما هنگامی که تعداد لایه‌های شبکه از یک حد مشخص فراتر می‌رود، عملکرد مدل به جای بهبود، دچار افت شدید می‌شود که به آن پدیده تخریب (\lr{Degradation}) گفته می‌شود. در این حالت نه‌تنها فرآیند آموزش سخت می‌شود، بلکه جزئیات مهم تصویر در اعماق شبکه محو یا دچار تغییرات مخرب می‌شوند.

معماری \lr{ResNet} برای حل این مشکل از مفهوم اتصالات میانبر استفاده می‌کند. \label{mianbor} در هر بلوک \lr{ResNet} یک مسیر میانبر اضافه می‌شود تا داده‌های ورودی یک بلوک بدون تغییر توسط لایه‌های کانولوشنی، مستقیماً به انتهای بلوک منتقل شوند. در پایان بلوک، اطلاعات مسیر میانبر با خروجی لایه‌های کانولوشنی ادغام می‌شود. از آنجا که ویژگی‌های ورودی از طریق مسیر میانبر حفظ می‌شوند، لایه‌های داخلی نیازی به بازسازی کامل ویژگی‌ها ندارند که این امر باعث کاهش بار محاسباتی و تسهیل فرآیند بهینه‌سازی می‌شود.

\paragraph{بلوک‌های سازنده و ساختار داخلی \lr{ResNet18}} \label{resnet block}

معماری \lr{ResNet18} از بلوک‌های ساده‌تری به نام \lr{BasicBlock} استفاده می‌کند.

هر \lr{BasicBlock} شامل موارد زیر است:
\begin{itemize}
	\item دو لایه کانولوشن با فیلترهای \lr{3$\times$3}
	\item لایه \lr{BatchNormalization} بعد از هر کانولوشن برای پایدارسازی آموزش
	\item تابع فعال‌سازی \lr{ReLU}
	\item یک مسیر میانبر که ورودی را به خروجی جمع می‌کند
\end{itemize}

معماری \lr{ResNet18} دارای \lr{18} لایه دارای وزن است.

\paragraph{پیش‌پردازش داده‌ها}

همان‌طور که در ساختار داخلی \lr{ResNet18} اشاره شد، این مدل داده‌ها را در قالب تصاویر \lr{3} کاناله با ابعاد \lr{224$\times$224} دریافت می‌کند. در حالی که داده‌های ما تصاویر تک کاناله \lr{28$\times$28} هستند. از طرفی تصاویر \lr{ImageNet} که \lr{ResNet18} با آن آموزش دیده است دارای میانگین و انحراف معیار مشخصی هستند.

بنابراین در مرحله پیش‌پردازش، تصاویر \lr{BreastMNIST} ابتدا باید به تصاویر \lr{3} کاناله با ابعاد \lr{224$\times$224} تبدیل شوند. همچنین چون از وزن‌های پیش‌آموزش‌دیده \lr{ResNet18} استفاده می‌کنیم، داده‌ها باید مطابق میانگین و انحراف معیار \lr{ImageNet} نرمال‌سازی شوند.

\paragraph{۱. ساختار شبکه مولد (\lr{Generator})}
ورودی این شبکه شامل یک بردار نویز $z \in \mathbb{R}^{100}$ و برچسب کلاس $y$ است. برچسب کلاس ابتدا از یک لایه جایگذاری (\lr{Embedding}) عبور کرده و به یک بردار ۱۰۰ بعدی تبدیل می‌شود. با اتصال (\lr{Concatenation}) این دو بردار، یک ورودی با ۲۰۰ کانال به شکل فضایی $200 \times 1 \times 1$ شکل می‌گیرد که وارد لایه‌های کانولوشن ترانهاده (\lr{ConvTranspose2d}) می‌شود:

\begin{itemize}
	\item \textbf{لایه اول:} ابعاد از $1 \times 1$ به $4 \times 4$ با ۵۱۲ کانال بسط می‌یابد (هسته ۴، گام ۱). فعال‌ساز: \lr{BatchNorm} و \lr{ReLU}.
	\item \textbf{لایه دوم:} ابعاد به $8 \times 8$ با ۲۵۶ کانال افزایش می‌یابد (هسته ۴، گام ۲، حاشیه ۱). فعال‌ساز: \lr{BatchNorm} و \lr{ReLU}.
	\item \textbf{لایه سوم:} ابعاد به $16 \times 16$ با ۱۲۸ کانال می‌رسد (هسته ۴، گام ۲، حاشیه ۱). فعال‌ساز: \lr{BatchNorm} و \lr{ReLU}.
	\item \textbf{لایه چهارم (خروجی):} تبدیل نهایی به ابعاد تصویر هدف یعنی $32 \times 32$ با ۳ کانال رنگی. تابع فعال‌ساز نهایی در این لایه \lr{Tanh} است که خروجی را در بازه $[-1, 1]$ تثبیت می‌کند.
\end{itemize}

\paragraph{۲. ساختار شبکه متمایزگر (\lr{Discriminator})}
شبکه متمایزگر باید تصویر (واقعی یا تولید شده) را به همراه برچسب آن ارزیابی کند. برای این منظور، برچسب کلاس ابتدا از یک لایه جایگذاری عبور کرده و به یک بردار ۱۰۲۴ بعدی تبدیل می‌شود. سپس این بردار به یک نقشه فضایی (\lr{Label Map}) با ابعاد $1 \times 32 \times 32$ تغییر شکل (\lr{Reshape}) یافته و به عنوان کانال چهارم به تصویر اصلی ($3 \times 32 \times 32$) متصل می‌گردد. ورودی نهایی با ابعاد $4 \times 32 \times 32$ وارد لایه‌های کانولوشن (\lr{Conv2d}) می‌شود:

\begin{itemize}
	\item \textbf{لایه اول:} کاهش ابعاد به $16 \times 16$ با ۶۴ کانال (هسته ۴، گام ۲، حاشیه ۱). فعال‌ساز: \lr{LeakyReLU(0.2)}. (بدون نرمال‌سازی دسته‌ای).
	\item \textbf{لایه دوم:} کاهش ابعاد به $8 \times 8$ با ۱۲۸ کانال. فعال‌ساز: \lr{BatchNorm} و \lr{LeakyReLU(0.2)}.
	\item \textbf{لایه سوم:} کاهش ابعاد به $4 \times 4$ با ۲۵۶ کانال. فعال‌ساز: \lr{BatchNorm} و \lr{LeakyReLU(0.2)}.
	\item \textbf{لایه چهارم (خروجی):} فشرده‌سازی به ابعاد $1 \times 1 \times 1$ (هسته ۴، گام ۱). خروجی این لایه از یک تابع \lr{Sigmoid} عبور می‌کند تا احتمال واقعی بودن تصویر را در بازه $[0, 1]$ نشان دهد.
\end{itemize}

\paragraph{۳. توابع زیان و بهینه‌سازی}
در این معماری از تابع زیان آنتروپی متقاطع باینری (\lr{BCE Loss}) استفاده شده است که فرمول پایه آن به صورت $L = -[y\log(p)+(1-y)\log(1-p)]$ تعریف می‌شود. توابع زیان برای هر دو شبکه به شکل زیر صورت‌بندی شده‌اند:

\begin{equation}
	L_D = \frac{1}{2} \Big[ BCELoss\big(D(x,y), 1\big) + BCELoss\big(D(G(z,y), y), 0\big) \Big]
\end{equation}
\begin{equation}
	L_G = BCELoss\big(D(G(z,y), y), 1\big)
\end{equation}

برای به‌روزرسانی وزن‌ها از الگوریتم \lr{Adam} با پارامترهای $\beta_1 = 0.5$ و $\beta_2 = 0.999$ استفاده شده است. یک نکته فنی و حیاتی در این پیاده‌سازی، استفاده از نرخ یادگیری نامتقارن است؛ در ابتدا نرخ یادگیری هر دو برابر با (\lr{lr = 0.0002}) بود اما در ابتدای آموزش مقدار لاس مولد خیلی بالا بود و بالاتر میرفت؛ برای حل این مشکل رویکرد رو تغییر دادم، به طوری که نرخ یادگیری مولد (\lr{lr = 0.0002}) دقیقاً چهار برابر نرخ یادگیری متمایزگر (\lr{lr = 0.00005}) تنظیم شده است. این رویکرد تضمین می‌کند که متمایزگر در مراحل اولیه آموزش بیش از حد قدرتمند نشود و فرصت یادگیری را از مولد سلب نکند.

\begin{itemize}
	\item \textbf{بروز پدیده فروپاشی حالت (\lr{Mode Collapse}):} همان‌طور که در خروجی‌های تصویر \ref{fig:basic_generated} مشهود است، شبکه برای هر بردار نویز ورودی، دقیقاً یک خروجی یکسان و نویزی را در هر کلاس کپی کرده است. این وضعیت بحرانی که در ادبیات شبکه‌های متخاصم به عنوان «فروپاشی حالت» یا \lr{Mode Collapse} شناخته می‌شود، زمانی رخ می‌دهد که شبکه مولد از ایجاد تنوع عاجز مانده و روی یک الگوی واحد که موقتاً متمایزگر را فریب داده است، قفل می‌شود.
	
	\item \textbf{عدم تعادل مخرب در مکانیزم به‌روزرسانی و نرخ یادگیری:} در این معماری، نرخ یادگیری مولد چهار برابر متمایزگر تنظیم شد و همزمان، مولد در هر گام دو بار آموزش دید (\lr{Double Update}). اگرچه این برتری محاسباتی باعث شد در اپوک‌های ابتدایی خطای مولد بسیار پایین باشد (مثلاً \lr{G Loss: 0.2190} در اپوک اول) و متمایزگر موقتاً مغلوب شود، اما این سرعت بالا منجر به پرش‌های ناگهانی و خشن گرادیان شد. شبکه نتوانست به تعادل نش (\lr{Nash Equilibrium}) دست یابد و با گذر از اپوک ۲۰، متمایزگر توانست الگوهای تکراری و نویزی مولد را به راحتی شناسایی کند که نتیجه آن افت شدید خطای \lr{D} و انفجار خطای \lr{G} به اعداد بالای ۲.۵ بود.
	
	\item \textbf{گلوگاه فضایی و شروع از نقطه کور ($1 \times 1$):} یکی از بزرگترین چالش‌های معماری مولد در این مدل، تلاش برای استخراج مستقیم ویژگی‌های پیچیده تصویر از یک تنسور با ابعاد فضایی $1 \times 1$ بود. این رویکرد به مثابه خلق یک ساختار هندسی پیچیده از هیچ است. در واقع، شبکه فاقد یک بوم فضایی اولیه و منطقی  بود تا بتواند الگوهای بافتی تومور را به درستی روی آن بسط (\lr{Upsampling}) دهد و توسعه بافت را پایه‌ریزی کند.
	
	\item \textbf{پیچیدگی مازاد و اضافه‌بار پارامترها (\lr{Over-parameterization}):} استفاده از ۵۱۲ کانال در لایه‌های اولیه برای تولید یک تصویر بسیار کوچک $32 \times 32$، یک معماری به شدت غول‌پیکر و غیربهینه ایجاد کرد. این حجم عظیم و غیرضروری از پارامترها به جای کمک به تعمیم‌پذیری مدل (\lr{Generalization})، باعث شد شبکه به سرعت ظرفیت یادگیری خود را صرف حفظ کردن (\lr{Memorization}) کند و به دلیل عدم توانایی در مدیریت این فضای پارامتری وسیع، دچار ناپایداری شده و در دریایی از نویزهای رنگی (\lr{RGB Artifacts}) واگرا شود.
	
	\item \textbf{خطای استراتژیک در مقیاس‌دهی مصنوعی تصاویر (\lr{Artificial Upscaling Fallacy}):}
	تصاویر اصلی در مجموعه داده \lr{BreastMNIST} دارای ابعاد ذاتی ۲۸×۲۸ پیکسل هستند. در این معماری اولیه، با این پیش‌فرض که افزایش ابعاد به ۳۲×۳۲ ممکن است به استخراج ویژگی‌های بهتر (\lr{Feature Extraction}) کمک کند، تصاویر مقیاس‌دهی (\lr{Upsample}) شدند. 
	از منظر پردازش سیگنال، بزرگ‌نمایی یک تصویر فراتر از رزولوشن ذاتی آن، هیچ اطلاعات یا فرکانس فضایی جدیدی به آن اضافه نمی‌کند، بلکه تنها مصنوعات درون‌یابی (\lr{Interpolation Artifacts}) ایجاد می‌کند. در شبکه‌های \lr{GAN}، متمایزگر به شدت به این مصنوعات مصنوعی حساس است. در نتیجه، شبکه به جای تمرکز بر یادگیری بافت بیولوژیکی و مورفولوژی تومور، درگیر یادگیری و بازتولید نویزهای ناشی از درون‌یابی شد. این عدم تطابق بین فرکانس طبیعی داده‌ها و ابعاد تحمیلی، یکی از دلایل اصلی افت کیفیت بافت در خروجی‌ها است.
\end{itemize}

در نهایت، ترکیب این معماریِ بیش از حد سنگین، ایجاد گلوگاه در بستر فضاییِ اولیه و اعمال فشار نامتعادل در فرآیند آموزش، به تولید مدلی منجر شد که از نظر تئوری و عملی قادر به حفظ ثبات متخاصم و سنتز تصاویر معنادار پزشکی نبود.


\newpage
\subsubsection{بررسی ساختار شبکه بهبود یافته (رویکرد میانی)}

بر مبنای تحلیل‌های آسیب‌شناختی صورت‌گرفته روی معماری اولیه، مشخص گردید که ریشه بسیاری از ناپایداری‌ها و واگرایی‌های شبکه در اضافه‌بار پارامتری، فقدان بستر فضایی اولیه و عدم تقارن در پویایی یادگیری نهفته است. از این رو، در فاز دوم توسعه، یک بازنگری بنیادین در ساختار شبکه‌ها و استراتژی آموزش اعمال شد. هدف این مرحله، دستیابی به یک معماری چابک‌تر (\lr{Lightweight}) و بهره‌گیری از تکنیک‌های پایدارسازی تئوریک بود.

\paragraph{۱. بازطراحی معماری شبکه: ایجاد بوم فضایی و بازگشت به ابعاد ذاتی}
نخستین و مهم‌ترین اصلاح ساختاری، تغییر رویکرد در نقطه شروع شبکه مولد بود. به جای تلاش بی‌ثمر برای استخراج ویژگی از یک نقطه فضایی کور ($1 \times 1$)، یک بستر فضایی اولیه (\lr{Spatial Canvas}) در شبکه تعریف شد. 

در این ساختار جدید، بردار نهانِ ترکیب‌شده ابتدا از طریق یک لایه تمام‌متصل به یک نقشه ویژگی با ابعاد $7 \times 7 \times 128$ نگاشت می‌شود. این ابعاد اولیه به شبکه اجازه می‌دهد تا الگوهای پایه و فرکانس‌پایینِ بافت تومور را پیش از عملیات مقیاس‌دهی (\lr{Upsampling}) پایه‌ریزی کند. فرآیند بسط فضایی در شبکه مولدِ بهبود یافته، مسیر زیر را طی می‌کند:
\begin{equation}
	(128 \times 7 \times 7) \xrightarrow{Upsample + Conv2d} (64 \times 14 \times 14) \xrightarrow{Conv2d} (32 \times 14 \times 14) \xrightarrow{Upsample + Conv2d} (1 \times 28 \times 28)
\end{equation}

این تغییرات سه دستاورد مهم به همراه داشت:
\begin{itemize}
	\item \textbf{لایه پردازش میانی ($14 \times 14 \times 32$):} افزودن یک لایه میانی بدون تغییر ابعاد فضایی، ظرفیت پردازشی (\lr{Capacity}) مولد را برای یادگیری بافت‌های پیچیده‌تر افزایش داد.
	\item \textbf{تناسب پارامتری (\lr{Right-sizing}):} کاهش شدید تعداد کانال‌ها (از ۵۱۲ به ۱۲۸ در لایه اول) معماری را با حجم محدود داده‌های \lr{BreastMNIST} متناسب ساخت و از پدیده بیش‌برازش ساختاری جلوگیری کرد.
	\item \textbf{بازگشت به فضای تک‌کاناله (\lr{Grayscale}):} با تنظیم کانال خروجی روی ۱ و ابعاد روی ۲۸×۲۸، شبکه دیگر درگیر تولید نویزهای رنگیِ بی‌معنی نشده و تمام ظرفیت خود را معطوف به یادگیری توزیع طیف خاکستری سونوگرافی نمود.
\end{itemize}
شبکه متمایزگر نیز به طور کاملاً متقارن ساده‌سازی شد تا در برابر این مولد چابک، رفتاری متعادل‌تر از خود نشان دهد.

\paragraph{۲. بکارگیری تکنیک هموارسازی برچسب‌ها (\lr{Label Smoothing})}
یکی از چالش‌های بنیادین در آموزش شبکه‌های \lr{GAN}، پدیده اعتماد به نفس کاذب (\lr{Overconfidence}) در متمایزگر است. در طبقه‌بندی باینری استاندارد، برچسب داده‌های واقعی به صورت مطلق برابر با $1.0$ و داده‌های جعلی برابر با $0.0$ در نظر گرفته می‌شود. از منظر ریاضیاتی تابع فعال‌سازی \lr{Sigmoid}، برای آنکه شبکه به خروجی دقیقاً $1.0$ دست یابد، باید وزن‌های لایه نهایی خود را به سمت بی‌نهایت میل دهد. این امر موجب می‌شود متمایزگر به جای یادگیری ویژگی‌های تعمیم‌پذیر، کوچکترین نویزهای داده‌های آموزش را حفظ کرده و به شدت مغرور شود.

زمانی که متمایزگر به اطمینان ۱۰۰ درصدی می‌رسد، تابع زیان متخاصم به اشباع رسیده و مشتق آن (گرادیان) به صفر میل می‌کند. در این حالت، پدیده مرگ گرادیان (\lr{Vanishing Gradient}) رخ داده و شبکه مولد هیچ سیگنال مفیدی برای اصلاح خطاهای خود دریافت نمی‌کند.

برای رفع این بحران، تکنیک \lr{Label Smoothing} به شبکه تزریق شد. در این روش، به جای استفاده از اهداف مطلق (سخت)، از اهداف نرم استفاده می‌شود:
$$y_{real} = 0.9 \quad , \quad y_{fake} = 0.0$$
با تنظیم آستانه برچسب‌های واقعی روی $0.9$، ما عملاً یک قانون اجباری برای متمایزگر وضع کردیم: \textit{«هیچ‌گاه حق نداری با قطعیت ۱۰۰ درصد تصمیم‌گیری کنی؛ همواره حداقل ۱۰ درصد احتمال خطا (\lr{Uncertainty}) برای خود در نظر بگیر.»}

این مداخله ریاضی تأثیرات شگرفی در پویایی مدل داشت:
\begin{enumerate}
	\item \textbf{جلوگیری از اشباع تابع زیان:} تابع \lr{Cross-Entropy} با هدف $0.9$ هرگز به صفر مطلق نمی‌رسد، بنابراین جریان گرادیان همواره زنده مانده و به طور پیوسته به سمت مولد روانه می‌شود.
	\item \textbf{کاهش جریمه‌های خشن:} زمانی که متمایزگر فروتن باشد، در صورت فریب خوردن از سوی مولد، جریمه‌های ناگهانی و پرش‌های شدید در نمودار \lr{Loss} ایجاد نمی‌کند، که این امر ثبات آموزش را به طرز چشمگیری افزایش می‌دهد.
\end{enumerate}

\paragraph{جمع‌بندی دستاوردهای معماری میانی}
بکارگیری همزمان معماری چابک‌سازی‌شده مبتنی بر بوم فضایی $7 \times 7$ و تکنیک ریاضی \lr{Label Smoothing}، توانست تا حد قابل قبولی جلوی واگرایی‌های خشن مدل اولیه را بگیرد و کیفیت بصری بافت‌ها را بهبود بخشد. با این وجود، این ساختار تنها یک بهبود \textit{نسبی} بود. اگرچه پایداری نسبی حاصل شد، اما مشکل جدیدتری داریم، و اون هم این هست که مدل داده های تکراری تولید میکند.

در گام نخست ارزیابی، بررسی پویایی توابع زیان (تصویر \ref{fig:mid_loss}) نشان‌دهنده یک پیشرفت چشمگیر و موفقیت‌آمیز در پایدارسازی ریاضی شبکه است. برخلاف معماری اولیه که در آن شاهد پرش‌های خشن، نوسانات انفجاری و مرگ گرادیان بودیم ، در این ساختار جدید، دینامیک یادگیری به یک ثبات بی‌نظیر دست یافته است:
\begin{itemize}
	\item \textbf{مهار متمایزگر:} به لطف استفاده از تکنیک هموارسازی برچسب‌ها (\lr{Label Smoothing = 0.9})، خطای متمایزگر (\lr{D Loss}) در یک بازه بسیار باریک و ایده‌آل بین $0.65$ تا $0.72$ نوسان می‌کند. از نظر تئوری ریاضی شبکه‌های \lr{GAN}، نقطه تعادل نش (\lr{Nash Equilibrium}) زمانی رخ می‌دهد که متمایزگر کاملاً گیج شده و احتمال $0.5$ را خروجی دهد. در تابع \lr{BCE Loss}، مقدار $-\ln(0.5) \approx 0.69$ است. نوسان خطای متمایزگر دقیقاً حول این عدد، اثبات می‌کند که شبکه از نظر ریاضی در ایده‌آل‌ترین حالت تئوریک خود قرار دارد.
	\item \textbf{تثبیت خطای مولد:} خطای مولد (\lr{G Loss}) نیز از رفتارهای واگرایانه فاصله گرفته و با ثباتی منطقی در بازه $0.80$ تا $0.95$ حرکت می‌کند. این امر نشان می‌دهد که مولد به طور پیوسته سیگنال‌های گرادیان مفیدی دریافت کرده و در حال یادگیری است.
\end{itemize}

با این وجود، در دنیای شبکه‌های متخاصم، زیبایی منحنی‌های زیان لزوماً به معنای تولید تصاویر باکیفیت نیست. ارزیابی خروجی‌های بصری شبکه پرده از چالش‌های جدیدی برمی‌دارد.

همان‌طور که در تصویر \ref{fig:mid_generated} مشاهده می‌شود، با وجود حل مشکل واگرایی و نویزهای رنگی (به دلیل بازگشت به فضای تک‌کاناله)، دو پدیده مخرب بافتی در خروجی‌ها به وضوح قابل تشخیص است:

\begin{enumerate}
	\item \textbf{بروز مجدد فروپاشی حالت (\lr{Mode Collapse}):}
	تصاویر تولید شده در هر کلاس (و حتی بین دو کلاس) شباهت بسیار زیادی به یکدیگر دارند و تنوع ساختاری (\lr{Diversity}) در آن‌ها به شدت پایین است. دلیل این امر، یافتن یک «نقطه امن محلی» (\lr{Safe Local Minimum}) توسط مولد است. از آنجا که متمایزگر به دلیل \lr{Label Smoothing} تا حدی ملایم و فروتن شده است، مولد متوجه شده که با تولید یک الگوی بافتیِ مبهم اما ثابت، می‌تواند متمایزگر را فریب دهد. در نتیجه، مولد بردار نویز ورودی ($z$) را عملاً نادیده گرفته (\lr{Latent Code Ignoring}) و از ایجاد تنوع بافتی که نیازمند ریسک‌پذیری در فضای بهینه‌سازی است، اجتناب می‌کند.
	
	\item \textbf{ظهور مصنوعات شطرنجی (\lr{Checkerboard Artifacts}):}
	بافت تصاویر تولید شده صاف و طبیعی نیست، بلکه دارای یک شبکه توری‌شکل و پیکسلی است که در ادبیات پردازش تصویر به آن مصنوعات شطرنجی گفته می‌شود. این پدیده یکی از مشکلات کلاسیک در شبکه‌های مولد است و ریشه در مکانیزم مقیاس‌دهی فضایی (\lr{Upsampling}) دارد. زمانی که ابعاد نقشه ویژگی از $7 \times 7$ به $14 \times 14$ و سپس $28 \times 28$ گسترش می‌یابد، اگر همپوشانی فیلترهای کانولوشنی (به‌ویژه در \lr{ConvTranspose2d} یا ترکیبات درون‌یابی) به صورت یکنواخت و متقارن توزیع نشود، برخی پیکسل‌ها بیش از حد فعال شده و فرکانس‌های بالای متناوب (شکل شطرنجی) را در تصویر حک می‌کنند. این مصنوعات برای شبکه‌های دسته‌بندی‌کننده نهایی (مانند \lr{ResNet}) به شدت گمراه‌کننده هستند، زیرا شبکه ممکن است این الگوهای شطرنجی را به اشتباه به عنوان ویژگی‌های بافت تومور (\lr{Biomarkers}) تفسیر کند.
\end{enumerate}

در یک جمع‌بندی، می‌توان گفت معماری میانی توانست مشکل «پایداری متخاصم» را در سطح گرادیان‌ها حل کند، اما در سطح بازنمایی بصری (\lr{Visual Representation}) گرفتار حداقل‌های محلی و مصنوعات ساختاری شد. رفع این مشکلات نیازمند تزریق تنوع به مولد (نظیر استفاده از \lr{Dropout}) و بکارگیری تکنیک‌های پیشرفته‌تر نرمال‌سازی مانند نرمال‌سازی طیفی (\lr{Spectral Normalization}) در متمایزگر است تا بدون نیاز به هموارسازی افراطی برچسب‌ها، پایداری و کیفیت تصویر به صورت توأمان تضمین گردد.

پایدارسازی و ارتقای این شبکه مدیون شش اصلاح ساختاری و ریاضیاتی است که در ادامه به تفصیل مورد تحلیل قرار می‌گیرند:

\paragraph{۱. جایگزینی \lr{BatchNorm} با نرمال‌سازی طیفی (\lr{Spectral Normalization})}
در شبکه‌های \lr{GAN}، استفاده از نرمال‌سازی دسته‌ای (\lr{Batch Normalization}) در متمایزگر باعث می‌شود خروجی شبکه برای یک تصویر، به سایر تصاویر موجود در آن دسته (\lr{Mini-batch}) وابسته شود. این وابستگی، تخمین دقیق توزیع داده‌ها را مختل می‌کند. برای حل این مشکل و همچنین جلوگیری از انفجار گرادیان، در شبکه نهایی تمامی لایه‌های \lr{BatchNorm} متمایزگر با نرمال‌سازی طیفی (\lr{Spectral Normalization}) جایگزین شدند.
از منظر ریاضی، نرمال‌سازی طیفی ماتریس وزن‌های هر لایه را بر بزرگترین مقدار منفرد آن (شعاع طیفی) تقسیم می‌کند و بدین ترتیب، شبکه را مجبور می‌سازد تا شرط پیوستگی لیپشیتز (با کران ثابت $K=1$) را رعایت کند:
$$ \| f(x) - f(y) \| \le 1 \cdot \| x - y \| $$
این محدودیت سخت‌گیرانه ریاضی باعث می‌شود متمایزگر تغییرات ناگهانی و خشن نداشته باشد و گرادیان‌هایی بسیار نرم، پیوسته و قابل‌اعتماد به سمت مولد ارسال کند.

\paragraph{۲. تداخل تنظیمی و حذف هموارسازی برچسب‌ها (\lr{Label Smoothing Removal})}
در معماری میانی، از تکنیک \lr{Label Smoothing} (تنظیم تارگت روی $0.9$) برای جلوگیری از اعتماد به نفس افراطی متمایزگر استفاده شده بود. اما در معماری نهایی، این تکنیک به طور کامل حذف گردید.
دلیل این امر یک \textbf{تداخل ساختاری} است: نرمال‌سازی طیفی (\lr{Spectral Norm}) به خودیِ خود با محدود کردن کران لیپشیتز، قدرت و اعتماد به نفس متمایزگر را کنترل و مهار می‌کند. اعمال همزمان \lr{Label Smoothing} در کنار \lr{Spectral Norm}، متمایزگر را بیش از حد تضعیف کرده و باعث می‌شود مولد به راحتی بر آن غلبه کند که نتیجه‌ی آن واگرایی مدل است. بنابراین، با اتکا به قدرت نرمال‌سازی طیفی، اهداف متمایزگر به همان حالت خالص (\lr{Hard Labels} یعنی $1.0$ و $0.0$) بازگردانده شدند.

\paragraph{۳. کنترل واگرایی با قاعده به‌روزرسانی دو مقیاسی (\lr{TTUR})}
با وجود اصلاحات فوق، در فازهای طولانی‌تر آموزش مشاهده شد که توابع زیان مولد و متمایزگر پس از مدتی از یکدیگر فاصله گرفته و دچار واگرایی می‌شوند. برای حل این چالش، از تکنیک قاعده به‌روزرسانی با دو مقیاس زمانی (\lr{Two Time-Scale Update Rule - TTUR}) بهره گرفته شد.
در این تکنیک، به جای استفاده از نرخ یادگیری یکسان، نرخ یادگیری متمایزگر (\lr{0.0003}) مقداری بیشتر از مولد (\lr{0.0002}) تنظیم شد. این رویکرد نامتقارن، تضمین می‌کند که متمایزگر همواره یک قدم از مولد جلوتر بوده و در حالت بهینه محلی خود باقی بماند تا بتواند سیگنال‌های خطای دقیق‌تری برای راهنمایی مولد صادر کند.

\paragraph{۴. اعمال دراپ‌اوت نامتقارن (\lr{Asymmetric Dropout})}
برای جلوگیری از حفظ کردن داده‌های آموزش توسط متمایزگر (بیش‌برازش)، یک لایه \lr{Dropout} با نرخ $0.3$ به لایه‌های پیچشی آن اضافه شد. 
اما این تکنیک \textbf{به صورت عمدی در شبکه مولد استفاده نشد}. دلیل این تصمیم‌گیری آن است که شبکه مولد برای تولید تصاویری با بافت طبیعی (به ویژه در تصاویر پزشکی که جزئیات بافت‌شناسی اهمیت حیاتی دارند)، نیازمند حفظ پیوستگی فضایی (\lr{Spatial Coherence}) است. خاموش کردن تصادفی نورون‌ها در مولد باعث ایجاد اختلال در جریان تولید تصویر شده و خروجی‌هایی تار، نویزی و فاقد انسجام هندسی تولید می‌کند.

\paragraph{۵. مقداردهی اولیه استاندارد و کنترل‌شده (\lr{Custom Weight Initialization})}
چارچوب \lr{PyTorch} به صورت پیش‌فرض از روش‌هایی نظیر کایمینگ (\lr{Kaiming/He}) برای مقداردهی اولیه وزن‌ها استفاده می‌کند که برای شبکه‌های دسته‌بندی عمیق بهینه شده‌اند، اما شبکه‌های \lr{GAN} به مقیاس وزن‌های اولیه به شدت حساس هستند. اگر به درستی وزن دهی انجام نشه ممکنه در همون ابتدای شبکه یکی از دیگری قوی تر باشه؛ برای رفع این مشکل، یک متد اختصاصی (\lr{\_weights\_init}) پیاده‌سازی شد:
\begin{itemize}
	\item \textbf{لایه‌های پیچشی (\lr{Conv Layers}):} وزن‌ها از یک توزیع نرمال با میانگین صفر و انحراف معیار \lr{۰.۰۲} استخراج می‌شوند: $W \sim \mathcal{N}(0, 0.02^2)$. این مقیاس کوچک از انفجار سیگنال در اولین پاس‌های پیش‌رو (\lr{Forward Pass}) جلوگیری می‌کند.
	\item \textbf{لایه‌های نرمال‌سازی (\lr{BatchNorm}):} برای جلوگیری از صفر شدن ویژگی‌ها، وزن‌های این لایه (پارامتر گاما) روی توزیع $\mathcal{N}(1, 0.02^2)$ و مقادیر بایاس (پارامتر بتا) دقیقاً روی عدد صفر ($0$) تنظیم شدند تا توزیع داده‌ها در ابتدا کاملاً دست‌نخورده باقی بماند.
\end{itemize}


\paragraph{۶. تزریق شرط و تقویت داده‌ها (\lr{Conditional Embedding \& Data Augmentation})}
در نهایت، برای تضمین پایداری و تعمیم‌پذیری بیشتر:
\begin{itemize}
	\item \textbf{تقویت داده‌ها:} در طول فرآیند آموزش، از تکنیک قرینه‌سازی افقی تصادفی (\lr{Random Horizontal Flipping}) استفاده شد تا تنوع بافتی تصاویری که به متمایزگر می‌رسند افزایش یابد.
	
\end{itemize}

\subsubsection{بررسی دقیق ساختار شبکه‌های نهایی}

در این بخش، با استناد به خروجی‌های تحلیل‌گر ساختاری (\lr{torchinfo})، معماری نهایی شبکه‌های مولد و متمایزگر کالبدشکافی می‌شود. این بررسی نشان می‌دهد که چگونه تغییرات اعمال شده توانسته‌اند با بهینه‌سازی جریان تنسورها و کاهش چشمگیر اضافه‌بار پارامتری، مدلی چابک و قدرتمند برای پردازش تصاویر پزشکی خلق کنند.

\paragraph{۱. شبکه مولد (\lr{Generator})}

همان‌طور که در تصویر \ref{fig:gen_arch_final} مشاهده می‌شود، معماری مولد با حدود\lr{ ۱.۳۴ }میلیون پارامتر آموزش‌پذیر (\lr{Trainable Params})، به گونه‌ای طراحی شده است که تعادلی بی‌نظیر میان ظرفیت یادگیری و جلوگیری از بیش‌برازش برقرار کند. جریان اطلاعات در این شبکه طی سه بلوک اصلی پردازش می‌شود:

\begin{itemize}
	\item \textbf{بلوک جایگذاری و ترکیب (\lr{Embedding \& Concatenation}):}
	شبکه در ابتدا یک دسته (\lr{Batch}) شامل ۳۲ بردار نویز ۱۰۰ بعدی ($32 \times 100$) و ۳۲ برچسب کلاس را دریافت می‌کند. برچسب‌ها با عبور از یک لایه جایگذاری به بردارهای ۱۰۰ بعدی تبدیل می‌شوند (با ۲۰۰ پارامتر تنظیم‌پذیر). با ترکیب این دو، ورودی بلوک بعدی به یک تنسور با ابعاد $32 \times 200$ ارتقا می‌یابد.
	
	\item \textbf{بلوک نگاشت فضایی مبنا (\lr{Dense Block}):}
	تنسور $32 \times 200$ وارد یک لایه تمام‌متصل خطی (\lr{Linear}) می‌شود که آن را به یک فضای ۶۲۷۲ بعدی بسط می‌دهد. این لایه با بیش از \lr{۱.۲ }میلیون پارامتر، سنگین‌ترین بخش مولد است و وظیفه دارد ارتباطات غیرخطی اولیه را شکل دهد. پس از عبور از لایه نرمال‌سازی دسته‌ای یک‌بعدی (\lr{BatchNorm1d}) و تابع \lr{ReLU}، این بردار مسطح به یک بوم فضایی سه‌بعدی با ابعاد $128 \times 7 \times 7$ تغییر شکل (\lr{Reshape}) می‌دهد.
	
	\item \textbf{بلوک بسط فضایی و سنتز (\lr{Convolutional Upsampling}):}
	یکی از مهم‌ترین اصلاحات این معماری، جایگزینی لایه‌های کانولوشن ترانهاده با ترکیب \lr{Upsample + Conv2d} است. این تغییر استراتژیک به طور کامل از بروز مصنوعات شطرنجی (\lr{Checkerboard Artifacts}) جلوگیری می‌کند. 
	در گام اول، ابعاد فضایی توسط لایه \lr{Upsample} دو برابر شده ($14 \times 14$) و سپس یک لایه کانولوشن استاندارد، عمق کانال‌ها را از ۱۲۸ به ۶۴ کاهش می‌دهد (همراه با \lr{BatchNorm2d} و \lr{ReLU}).
	در گام دوم، ابعاد فضایی مجدداً دو برابر شده ($28 \times 28$) و آخرین لایه کانولوشن، نقشه ویژگی را به یک تصویر تک‌کاناله نهایی با ابعاد $1 \times 28 \times 28$ تبدیل می‌کند. در نهایت، تابع \lr{Tanh} توزیع نوری پیکسل‌ها را در بازه استاندارد تنظیم می‌نماید.
\end{itemize}

بر اساس تصویر \ref{fig:disc_arch_final}، شبکه متمایزگر به صورت یک مدل بسیار سبک و کارآمد با تنها \lr{۱۴۰,۹۶۱ }پارامتر آموزش‌پذیر طراحی شده است. این تفاوت حجم پارامتر (۱۴۰ هزار در برابر\lr{ ۱.۳۴ }میلیون در مولد) یک رویکرد هوشمندانه برای جلوگیری از غلبه زودهنگام متمایزگر است. جریان ارزیابی در این شبکه به شرح زیر است:

\begin{itemize}
	\item \textbf{بلوک هم‌آمیزی شرطی (\lr{Conditional Fusion}):}
	برچسب کلاس ورودی از طریق لایه جایگذاری به یک بردار ۷۸۴ بعدی تبدیل می‌شود. از آنجا که ابعاد تصویر ورودی $28 \times 28$ (معادل ۷۸۴ پیکسل) است، این بردار شرطی به یک نقشه فضایی $1 \times 28 \times 28$ تغییر شکل یافته و در کنار تصویر ورودی (تولیدی یا واقعی) قرار می‌گیرد. در نتیجه، تنسور ورودی متمایزگر ساختاری ۲ کاناله با ابعاد $2 \times 28 \times 28$ پیدا می‌کند.
	
	\item \textbf{بلوک استخراج ویژگی (\lr{Feature Extraction}):}
	این بلوک شامل دو لایه کانولوشن استاندارد با گام حرکت (\lr{Stride}) ۲ است که وظیفه کاهش ابعاد فضایی و استخراج الگوهای بافتی را بر عهده دارند.
	لایه اول، ابعاد را به $64 \times 14 \times 14$ تقلیل داده و لایه دوم آن را به $128 \times 7 \times 7$ می‌رساند. پس از هر لایه کانولوشن، از تابع فعال‌سازی \lr{LeakyReLU} برای حفظ گرادیان‌های منفی و از لایه \lr{Dropout} (برای کنترل بیش‌برازش و تضعیف عمدی متمایزگر جهت همگامی با مولد) استفاده شده است.
	
	\item \textbf{بلوک تصمیم‌گیری (\lr{Classification Block}):}
	نقشه ویژگی نهایی ($128 \times 7 \times 7$) توسط لایه \lr{Flatten} مسطح شده و به یک بردار ۶۲۷۲ بعدی تبدیل می‌گردد. در نهایت، یک لایه خطی (\lr{Linear}) با ۶۲۷۳ پارامتر، این بردار را به یک اسکالر خروجی متراکم می‌کند که با عبور از تابع \lr{Sigmoid}، احتمال واقعی یا جعلی بودن تصویر را تعیین می‌نماید.
\end{itemize}

در یک جمع‌بندی ساختاری، انتقال از معماری‌های غول‌پیکرِ مبتنی بر کانولوشن‌های ترانهاده به این معماری چابک و مبتنی بر \lr{Upsampling}، نه تنها هزینه‌های محاسباتی (نظیر \lr{Forward/Backward pass size} که در مجموع زیر ۱۵ مگابایت است) را به شدت کاهش داده، بلکه با ایجاد یک قیف اطلاعاتی منطقی، شبکه را در استخراج و بازسازی ویژگی‌های حیاتی سونوگرافی به حداکثر بهره‌وری رسانده است.

\subsubsection{بررسی فرایند آموزش و استراتژی ارزیابی شبکه}

فرآیند آموزش در شبکه‌های متخاصم مولد (\lr{GAN}) به دلیل ماهیت بازی کمینه‌بیشینه (\lr{Minimax Game}) همواره چالش‌برانگیز است. در این معماری، علاوه بر تنظیم دقیق دینامیک یادگیری بین مولد و متمایزگر، پروتکل‌های پیشرفته‌ای برای آماده‌سازی داده‌ها، ارزیابی پایداری و ذخیره‌سازی هوشمندِ وزن‌های شبکه پیاده‌سازی شده است که در ادامه به تفصیل مورد بررسی قرار می‌گیرند.

\paragraph{۱. پروتکل آماده‌سازی و تقویت داده‌ها (\lr{Data Augmentation \& Transformation})}
یکی از اصول حیاتی در ارزیابی مدل‌های یادگیری عمیق، تفکیک دقیق سیاست‌های پیش‌پردازش بین فاز آموزش (\lr{Train}) و فاز ارزیابی/آزمون (\lr{Validation/Test}) است. همان‌طور که در بلوک `data transform augment` و `data transform` مشخص است، این تفکیک به شکل زیر اعمال شده است:

\begin{itemize}
	\item \textbf{فاز آموزش (\lr{Training}):} داده‌ها ابتدا به فرمت تک‌کاناله (\lr{Grayscale}) و ابعاد $28 \times 28$ تبدیل شده، سپس یک تبدیل تصادفی قرینه‌سازی افقی (\lr{RandomHorizontalFlip}) با احتمال ۵۰ درصد روی آن‌ها اعمال می‌گردد. در نهایت، تصاویر به تنسور تبدیل شده و در بازه $[-1, 1]$ نرمال‌سازی می‌شوند (با میانگین و انحراف معیار $0.5$). این تقویت داده‌ها (\lr{Data Augmentation}) باعث افزایش تنوع الگوهای بافتی شده و از بیش‌برازش متمایزگر جلوگیری می‌کند.
	\item \textbf{فاز ارزیابی و آزمون (\lr{Validation/Test}):} در این فاز، تبدیل قرینه‌سازی تصادفی به صورت عمدی \textbf{حذف} شده است. دلیل این امر، الزام به حفظ ثبات در معیارهای ارزیابی است. اگر بر روی مجموعه اعتبارسنجی نیز تغییرات تصادفی اعمال می‌شد، خروجی مدل در هر اپوک با یک تصویر متفاوت ارزیابی می‌گردید که این مسئله نویز شدیدی به معیارهای سنجش وارد کرده و مقایسه عملکرد مدل در طول زمان را غیرممکن می‌ساخت.
\end{itemize}

علاوه بر این، در تنظیمات بارگذار داده آموزشی (`train loader`)، پارامتر `drop last=True` فعال شده است. از آنجا که در شبکه مولد از لایه‌های نرمال‌سازی دسته‌ای (\lr{Batch Normalization}) استفاده شده است، این پارامتر تضمین می‌کند که اگر در انتهای مجموعه داده، تعداد تصاویر باقیمانده کمتر از حجم دسته (مثلاً ۳۲) بود، آن دسته کوچک دور ریخته شود. دسته‌های کوچک باعث ایجاد ناپایداری شدید در محاسبات میانگین و واریانسِ لایه‌های \lr{BatchNorm} شده و می‌توانند کل فرآیند آموزش را دچار فروپاشی کنند.

\paragraph{۲. ساختار حلقه آموزش متخاصم}
در هر اپوک، شبکه بر اساس پروتکل استاندارد شبکه‌های متخاصم، در دو مرحله متوالی آموزش می‌بیند:

\begin{enumerate}
	\item \textbf{به‌روزرسانی متمایزگر (\lr{Train D}):} ابتدا متمایزگر با یک دسته از تصاویر واقعی و یک دسته از تصاویر تولید شده توسط مولد تغذیه می‌شود. هدف متمایزگر بهینه‌سازی تابع زیان زیر است تا توانایی تفکیک خود را به حداکثر برساند:
	$$ L_D = \frac{1}{2} \Big[ L_{real} + L_{fake} \Big] $$
	\item \textbf{به‌روزرسانی مولد (\lr{Train G}):} در گام بعدی، مولد با استفاده از یک بردار نویز جدید تصویر تولید می‌کند. هدف مولد فریب دادن متمایزگر است؛ بنابراین تابع زیان آن به گونه‌ای محاسبه می‌شود که خروجی متمایزگر $D(G(z,y))$ به سمت عدد ۱ (واقعی بودن) هدایت شود.
\end{enumerate}
پس از اتمام آموزش روی تمامی دسته‌ها، مدل وارد فاز اعتبارسنجی (\lr{Validation}) شده و معیار تعادل برای آن محاسبه می‌گردد.

\paragraph{۳. طراحی معیار ارزیابی اختصاصی: امتیاز تعادل (\lr{Balance Score})}
در شبکه‌های عصبی استاندارد (مانند طبقه‌بندها)، همواره مدلی که کمترین خطای اعتبارسنجی (\lr{Validation Loss}) را داشته باشد به عنوان بهترین مدل انتخاب می‌شود. اما در شبکه‌های \lr{GAN}، توابع زیان همواره در حال رقابت و نوسان هستند و کاهش خطای مولد لزوماً به معنای تولید تصاویر بهتر نیست. برای حل این چالش، یک تابع ارزیابی اختصاصی تحت عنوان `evaluate gan balance` توسعه یافت.

این تابع عملکرد شبکه را در برابر مفهوم تئوریک تعادل نش (\lr{Nash Equilibrium}) می‌سنجد. در این تابع، متمایزگر تصاویری را از مجموعه اعتبارسنجی (\lr{Real}) و تصاویری که به تازگی توسط مولد سنتز شده‌اند (\lr{Fake}) دریافت کرده و میانگین احتمالات پیش‌بینی شده را محاسبه می‌کند:
\begin{align*}
	P_{real} &= \frac{1}{N} \sum D(x) \\
	P_{fake} &= \frac{1}{N} \sum D(G(z, y))
\end{align*}

در یک شبکه \lr{GAN} که به ایده‌آل‌ترین حالت خود (تعادل نش) رسیده است، متمایزگر دیگر قادر به تفکیک داده‌های واقعی از جعلی نیست و برای هر دو گروه مقدار احتمال $0.5$ را بازمی‌گرداند. بر همین اساس، فرمول \lr{Balance Score} به شکل زیر تعریف شد:
\begin{equation}
	BalanceScore = |P_{real} - 0.5| + |P_{fake} - 0.5|
\end{equation}

\textbf{تفسیر ریاضی معیار:}
\begin{itemize}
	\item هرچه این امتیاز به \textbf{صفر} نزدیک‌تر باشد، شبکه در وضعیت متعادل‌تری قرار دارد.
	\item اگر شبکه دچار عدم تعادل شود (مثلاً متمایزگر بسیار قوی شده و $P_{real}=0.95$ و $P_{fake}=0.05$ شود)، امتیاز تعادل به شدت افزایش می‌یابد ($Score = 0.45 + 0.45 = 0.90$).
\end{itemize}

\paragraph{۴. استراتژی ذخیره‌سازی وضعیت (\lr{Checkpointing Strategy})}
بر خلاف رویکردهای سنتی، در این پیاده‌سازی، شرط ذخیره کردن «بهترین مدل» بر روی کمینه‌سازی خطای مولد استوار نیست، بلکه الگوریتم به دنبال یافتن نقطه‌ای است که در آن کمترین میزان امتیاز تعادل ثبت شود (\lr{Minimum Balance Score}). 

در تابع traincDCGAN، در انتهای هر اپوک یک دیکشنری جامع شامل شماره اپوک، ماتریس وزن‌های هر دو شبکه (\lr{State Dicts})، میانگین توابع زیان و امتیاز تعادل تشکیل شده و تحت سه فرمت استراتژیک در دیسک ذخیره می‌شود:
\begin{enumerate}
	\item \lr{\textbf{checkpoint\_epoch\_X.pth}:} ذخیره وضعیت دقیق هر اپوک، جهت فراهم آوردن امکان بازگشت (\lr{Rollback}) به هر نقطه دلخواه از تاریخچه آموزش.
	\item \lr{\textbf{last\_model.pth}:} یک فایل بازنویسی‌شونده (\lr{Overwriting}) که همواره وضعیت آخرین اپوک اجرا شده را در خود نگه می‌دارد و برای ادامه آموزش (\lr{Resume Training}) در صورت قطع فرآیند کاربرد دارد.
	\item \lr{\textbf{best\_model.pth}:} این فایل تنها زمانی بروزرسانی می‌شود که مدل فعلی توانسته باشد یک رکورد جدید در نزدیک شدن به تعادل نش (کاهش \lr{Balance Score}) ثبت کند. این دقیقاً همان مدلی است که در فاز نهایی برای استخراج ویژگی و تولید تصویر مورد استفاده قرار خواهد گرفت.
\end{enumerate}


\subsubsection{بررسی نتایج نهایی مدل}

پس از اعمال تغییرات بنیادین در معماری، بهینه‌سازی جریان تنسورها و تزریق تکنیک‌های پایدارسازی ریاضی، مدل نهایی به مدت ۵۰۰ اپوک تحت آموزش قرار گرفت. در این بخش، رفتار دینامیکی توابع زیان و کیفیت بصری تصاویر سنتز شده به عنوان خروجی‌های نهایی این چارچوب مورد تحلیل و ارزیابی قرار می‌گیرند.

\begin{itemize}
	\item \textbf{تثبیت در مجاورت تعادل نش (\lr{Nash Equilibrium}):} در ۱۰۰ اپوک ابتدایی، خطای هر دو شبکه حول مقدار $0.69$ تا $0.72$ نوسان می‌کند. در تابع زیان \lr{BCE}، زمانی که متمایزگر نتواند داده واقعی را از جعلی تشخیص دهد (احتمال $0.5$)، مقدار خطا برابر با $-\ln(0.5) \approx 0.693$ خواهد بود. این نوسانات اولیه نشان می‌دهد که شبکه از همان ابتدا در یک تعادل تئوریک ایده‌آل کار خود را آغاز کرده است.
	
	\item \textbf{تأثیر شگرف نرمال‌سازی طیفی (\lr{Spectral Normalization}):} بر خلاف مدل اولیه که شاهد پرش‌های انفجاری و مرگ گرادیان (خطای بالای ۲.۵) بودیم، در اینجا نمودار فاقد هرگونه پرش خشن است. محدودیت لیپشیتز که توسط نرمال‌سازی طیفی بر متمایزگر تحمیل شده، باعث شده است تا متمایزگر نتواند با قطعیت ۱۰۰ درصد تصمیم‌گیری کند. این امر یک جریان گرادیان (\lr{Gradient Flow}) پیوسته، نرم و بسیار غنی را در تمام طول ۵۰۰ اپوک برای مولد تضمین کرده است.
	
	\item \textbf{جدایی کنترل‌شده به واسطه قاعده \lr{TTUR}:} از اپوک ۲۰۰ به بعد، مشاهده می‌شود که خطای متمایزگر به آرامی در حال کاهش (میل به سمت $0.60$) و خطای مولد به تدریج در حال افزایش (میل به سمت $0.85$) است. این پدیده دقیقاً خروجیِ مطلوبِ تکنیک \lr{TTUR} (نرخ یادگیری متمایزگر $0.0003$ در برابر مولد $0.0002$) است. متمایزگر به دلیل نرخ یادگیری کمی بالاتر، همواره یک قدم از مولد جلوتر است و به مرور زمان برتری خفیفی پیدا می‌کند. این «برتری کنترل‌شده» بسیار حیاتی است؛ زیرا اگر متمایزگر ضعیف شود، نمی‌تواند ایرادات ریز تصاویر تولیدی را تشخیص دهد. در اینجا، متمایزگر به صورت آهسته سخت‌گیرتر می‌شود و مولد را مجبور می‌کند تا در اپوک‌های پایانی به جزئیات بسیار ظریف‌تری توجه کند، بدون آنکه منجر به فروپاشی شبکه شود.
\end{itemize}

\begin{enumerate}
	\item \textbf{درمان قطعی فروپاشی حالت (\lr{Mode Collapse Resolution}):}
	در این تصاویر، هیچ‌گونه شباهت مطلق یا کپی‌برداری در بین نمونه‌های تولید شده دیده نمی‌شود. هر یک از ۲۰ تصویرِ نمایش داده شده در هر کلاس، دارای مورفولوژی، توزیع نوری و شکل هندسی کاملاً منحصر‌به‌فردی است. این تنوع ساختاری (\lr{Structural Diversity}) اثبات می‌کند که شبکه مولد توانسته است به درستی کل فضای نهان ($z \in \mathbb{R}^{100}$) را به توزیع داده‌های واقعی نگاشت کند و در حداقل‌های محلی گرفتار نشده است.
	
	\item \textbf{حذف مصنوعات شطرنجی و رنگی (\lr{Artifact Elimination}):}
	جایگزینی لایه‌های کانولوشن ترانهاده با معماری مبتنی بر \lr{Upsample}، به طور کامل شبکه‌های توری‌شکل و مصنوعات شطرنجی را از بین برده است. بافت تصاویر، همانند سونوگرافی‌های واقعی، دارای نویزهای نقطه‌ای طبیعی (\lr{Speckle Noise}) و گرادیان‌های نوری پیوسته است. همچنین، بازگشت به فضای تک‌کاناله (\lr{Grayscale}) باعث تمرکز کامل ظرفیت شبکه بر روی اکوژنیسیته (\lr{Echogenicity}) تومورها شده است.
	
	
\end{enumerate}

در نتیجه، معماری نهایی با مهار ریاضیاتیِ گرادیان‌ها و بهینه‌سازی جریان اطلاعات فضایی، توانسته است تصاویری تولید کند که نه تنها از نظر تنوع و کیفیت بصری در سطح بسیار بالایی قرار دارند، بلکه از نظر ویژگی‌های مورفولوژیکِ پزشکی نیز کاملاً با کلاسِ شرطیِ داده‌شده منطبق هستند.

\subsubsection{تولید 2000 داده‌های مصنوعی و توازن‌بخشی به مجموعه داده آموزش}

یکی از چالش‌های بنیادین در مجموعه داده‌های پزشکی نظیر \lr{BreastMNIST}، عدم توازن شدید کلاس‌ها (\lr{Class Imbalance}) است. در مجموعه داده آموزش اولیه، تعداد نمونه‌های بدخیم (۳۹۹ تصویر) به مراتب بیشتر از نمونه‌های خوش‌خیم (۱۴۷ تصویر) بود. این اختلاف فاحش باعث می‌شود مدل‌های دسته‌بند (نظیر \lr{ResNet}) دچار سوگیری (\lr{Bias}) به سمت کلاس اکثریت شوند. برای حل این بحران، از شبکه \lr{cDCGAN} نهایی به عنوان یک موتور تولید داده و از یک استراتژی توازن‌بخشی اولویت‌محور (\lr{Priority-based Balancing}) استفاده گردید.

\paragraph{۱. استراتژی تولید استخر داده‌ها (تولید ۲۰۰۰ نمونه)}
ابتدا با استفاده از مدل ، یک «استخر داده‌های مصنوعی» (\lr{Synthetic Data Pool}) ایجاد شد. به این منظور، شبکه مولد در حالت ارزیابی (\lr{eval}) قرار گرفت و با تزریق بردارهای نویز تصادفی از توزیع نرمال، تعداد ۲۰۰۰ تصویر مجزا برای کلاس خوش‌خیم و ۲۰۰۰ تصویر برای کلاس بدخیم تولید گردید. 

\paragraph{۲. استراتژی توازن‌بخشی اولویت‌محور (\lr{Target-Driven Balancing})}
پس از تولید داده‌ها، استراتژی ساخت مجموعه داده نهایی بر مبنای حفظ حداکثری اصالت داده‌ها بنا شد. در این روش، یک سقف هدف ($T=600$) برای هر کلاس در نظر گرفته شد. الگوریتم توازن‌بخشی به این صورت عمل می‌کند که ابتدا \textbf{تمامی داده‌های واقعی} موجود در آن کلاس را وارد مجموعه نهایی می‌کند و تنها برای پر کردن خلأ موجود تا رسیدن به سقف هدف، از داده‌های مصنوعی استفاده می‌نماید. رابطه ریاضی این استراتژی به شکل زیر است:
$$ N_{fake} = \max(0, T - N_{real}) $$
بر اساس لاگ‌های اجرایی، توزیع داده‌ها به شکل زیر مهندسی شد:
\begin{itemize}
	\item \textbf{کلاس ۰ (خوش‌خیم):} از آنجا که تنها ۱۴۷ تصویر واقعی وجود داشت، دقیقاً ۴۵۳ تصویر مصنوعی به صورت تصادفی از استخر داده‌ها انتخاب و به این کلاس تزریق شد تا به عدد ۶۰۰ برسد.
	\item \textbf{کلاس ۱ (بدخیم):} با وجود ۳۹۹ تصویر واقعی، تنها به ۲۰۱ تصویر مصنوعی نیاز بود تا این کلاس نیز دقیقاً به سقف ۶۰۰ نمونه برسد.
\end{itemize}
در نهایت، یک مجموعه داده آموزشی کاملاً متوازن شامل ۱۲۰۰ نمونه (۶۰۰ خوش‌خیم و ۶۰۰ بدخیم) شکل گرفت.

\paragraph{۳. منطق محدودسازی تقویت داده‌ها صرفاً به فاز آموزش}
یکی از اصول تخطی‌ناپذیر در متدولوژی یادگیری ماشین بالینی، عدم دستکاری مجموعه داده‌های اعتبارسنجی (\lr{Validation}) و آزمون (\lr{Test}) است. تزریق داده‌های \lr{GAN} یا هرگونه توازن‌بخشی \textbf{تنها و تنها روی مجموعه آموزش (\lr{Train})} اعمال شد. 
دلیل این امر آن است که مدل باید در یک محیط غنی و متوازن «آموزش» ببیند تا مرزهای تصمیم‌گیری (\lr{Decision Boundaries}) را بدون سوگیری یاد بگیرد؛ اما مرحله «ارزیابی» باید دقیقاً بازتاب‌دهنده‌ی واقعیت نامتوازنِ دنیای کلینیکال باشد. اگر داده‌های آزمون نیز با \lr{GAN} بالانس می‌شدند، معیارهای ارزیابی نهایی (\lr{Precision, Recall}) مخدوش شده و عملکرد مدل در دنیای واقعی به صورت کاذب، ایده‌آل به نظر می‌رسید (\lr{Data Leakage}).

\subsubsection{ارزیابی نهایی: تحلیل عملکرد طبقه‌بند \lr{ResNet} (داده‌های اصیل در برابر داده‌های افزوده با \lr{GAN})}

\paragraph{۱. دستاورد بالینی: افزایش حساسیت در تشخیص سرطان (کلاس بدخیم)}
مهم‌ترین و حیاتی‌ترین شاخص در غربالگری تومورها، شاخص \lr{Recall} (یا حساسیت - \lr{Sensitivity}) در کلاس بدخیم است. این شاخص نشان می‌دهد که مدل چند درصد از بیماران سرطانی واقعی را به درستی تشخیص داده است. 
همان‌طور که در جدول مشخص است، تزریق داده‌های \lr{GAN} باعث ارتقای \lr{Recall} کلاس بدخیم از $0.93$ به $0.95$ شده است. از منظر پزشکی، این یک موفقیت بسیار بزرگ محسوب می‌شود؛ زیرا خطای نوع دوم یا منفی کاذب (\lr{False Negative} - یعنی تشخیص تومور سرطانی به عنوان بافت سالم) هزینه‌ای معادل با به خطر افتادن جان بیمار دارد. داده‌های مصنوعی تولید شده توسط \lr{cDCGAN} با ایجاد تنوع بیشتر در مورفولوژی تومورهای بدخیم، به \lr{ResNet} کمک کردند تا الگوهای پنهان سرطان را بهتر بشناسد و بیماران سرطانیِ کمتری را از دست بدهد.

\paragraph{۲. مصالحه آماری (\lr{Trade-off}): افت \lr{Recall} در کلاس خوش‌خیم}
افزایش حساسیت به تومورهای بدخیم همواره با یک مصالحه (\lr{Trade-off}) همراه است. در مدل متوازن‌شده، شاخص \lr{Recall} برای کلاس خوش‌خیم از $0.69$ به $0.57$ افت کرده است. این بدان معناست که مدل در برابر توده‌های خوش‌خیم به شدت محتاط و بدبین شده است و تمایل دارد موارد مشکوک را «بدخیم» اعلام کند. 
ریشه این رفتار در توزیع داده‌های مصنوعی نهفته است. از آنجا که برای رسیدن به مرز ۶۰۰ نمونه، ۴۵۳ نمونه مصنوعی به کلاس خوش‌خیم (نسبت به تنها ۲۰۱ نمونه بدخیم مصنوعی) افزوده شد، حجم عظیمی از ویژگی‌های سنتز شده وارد فضای این کلاس گردید. به نظر می‌رسد همپوشانی لبه‌های تصمیم‌گیری (\lr{Decision Boundary Overlap}) در داده‌های مصنوعی خوش‌خیم با داده‌های بدخیم، باعث شده تا طبقه‌بند نهایی (\lr{ResNet}) آستانه تصمیم‌گیری خود را به نفع کلاس بدخیم تغییر دهد.

\paragraph{۳. ارتقای دقت پیش‌بینی کلاس خوش‌خیم}
نکته جالب توجه دیگر، افزایش \lr{Precision} کلاس خوش‌خیم از $0.78$ به $0.80$ در مدل متوازن است. این یعنی زمانی که مدل جدیدِ ما توده‌ای را «خوش‌خیم» اعلام می‌کند، با اطمینان و قطعیت بسیار بیشتری می‌توان به این تشخیص اعتماد کرد، زیرا هشدار کاذب کمتری در این جهت صادر می‌کند.

\paragraph{جمع‌بندی}
در ارزیابی نهایی، اگرچه تزریق داده‌های مصنوعی تولید شده توسط \lr{cDCGAN} باعث کاهش ۲ درصدی در دقت کلی (\lr{Accuracy}) مدل گردید، اما این رویکرد به صورت هدفمند، سوگیری (\lr{Bias}) مدل را به نفع \textbf{«تشخیص ایمن‌تر»} تغییر داد. در کاربردهای کلینیکال، مدلی که \lr{Recall} سرطان در آن ۹۵ درصد باشد اما هشدارهای کاذب (\lr{False Positives}) بیشتری تولید کند (مدل مبتنی بر \lr{GAN})، به مراتب ارزشمندتر و قابل‌اعتمادتر از مدلی است که دقت کلی بالاتری دارد اما بیماران سرطانی بیشتری را از دست می‌دهد (مدل نامتوازن اولیه). بنابراین، این آزمایش پتانسیل بالای شبکه‌های متخاصم را در بازطراحی مرزهای تصمیم‌گیری به نفع پروتکل‌های ایمنی پزشکی به اثبات می‌رساند.

PCA یک روش \textbf{خطی} برای کاهش ابعاد است که تلاش می‌کند داده‌ها را به فضایی با ابعاد کمتر منتقل کند به‌گونه‌ای که بیشترین میزان واریانس حفظ شود. مراحل اصلی عبارتند از:

1. استانداردسازی داده‌ها  
2. محاسبه ماتریس کوواریانس  
3. محاسبه مقادیر ویژه و بردارهای ویژه  
4. انتخاب مؤلفه‌های اساسی و تصویربرداری داده‌ها روی آن‌ها  

\paragraph{مزایا}
\begin{itemize}
	\item سادگی و سرعت بالا  
	\item کاهش نویز  
	\item عدم بروز بیش‌برازش (Overfitting)  
\end{itemize}

\paragraph{معایب}
\begin{itemize}
	\item محدودیت در نمایش ساختارهای غیرخطی  
	\item حساسیت به نقاط پرت  
	\item دشوار شدن تفسیر مؤلفه‌ها  
\end{itemize}

\subsubsection{الگوریتم \lr{ISOMAP (Isometric Mapping)}}

\paragraph{شیوه عملکرد}
ISOMAP یک روش \textbf{غیرخطی} مبتنی بر خمینه‌ها است و به‌جای فاصله اقلیدسی، فاصله ژئودزیک را حفظ می‌کند.

\paragraph{مراحل اجرای الگوریتم \lr{ISOMAP}}

\paragraph{۱. تشکیل گراف همسایگی}
در این مرحله ساختار محلی داده‌ها مدل می‌شود. فرض اصلی ISOMAP این است که داده‌های با بعد بالا روی یک \textit{خمینه} (Manifold) با بعد کمتر قرار دارند. برای تقریب این خمینه، یک گراف همسایگی ساخته می‌شود که در آن هر نقطه داده یک رأس (Node) در گراف است.

برای تعیین همسایه‌ها معمولاً از دو روش استفاده می‌شود:

\begin{itemize}
	\item \lr{K-Nearest Neighbors (KNN)}: برای هر نقطه، \(k\) تا از نزدیک‌ترین نقاط بر اساس فاصله اقلیدسی انتخاب می‌شوند و بین آن‌ها یال برقرار می‌شود.
	\item \lr{\(\epsilon\)-Neighborhood}: تمام نقاطی که فاصله اقلیدسی آن‌ها از یک نقطه کمتر از مقدار آستانه \(\epsilon\) باشد، به عنوان همسایه در نظر گرفته می‌شوند.
\end{itemize}

وزن هر یال در گراف برابر با فاصله اقلیدسی بین دو نقطه است:

\[
d_{ij} = \|x_i - x_j\|
\]

در نتیجه یک گراف وزن‌دار ساخته می‌شود که تنها ارتباطات محلی داده‌ها را حفظ می‌کند.

\paragraph{۲. محاسبه فواصل ژئودزیک}
پس از ساخت گراف همسایگی، هدف تخمین فاصله واقعی بین نقاط روی خمینه است. در فضای غیرخطی، فاصله مستقیم اقلیدسی بین دو نقطه دور ممکن است مسیر واقعی روی سطح خمیده را نشان ندهد. به همین دلیل ISOMAP از \textbf{فاصله ژئودزیک} استفاده می‌کند.

فاصله ژئودزیک بین دو نقطه به صورت کوتاه‌ترین مسیر روی گراف تعریف می‌شود. برای محاسبه این فاصله‌ها از الگوریتم‌های کوتاه‌ترین مسیر استفاده می‌شود:

\begin{itemize}
	\item الگوریتم \textbf{Dijkstra} برای محاسبه کوتاه‌ترین مسیر از یک گره به تمام گره‌های دیگر
	\item الگوریتم \textbf{Floyd–Warshall} برای محاسبه کوتاه‌ترین مسیر بین تمام جفت گره‌ها
\end{itemize}

خروجی این مرحله یک ماتریس فاصله ژئودزیک است:

\[
D_G =
\begin{bmatrix}
	0 & d_G(x_1,x_2) & \dots & d_G(x_1,x_n) \\
	d_G(x_2,x_1) & 0 & \dots & d_G(x_2,x_n) \\
	\vdots & \vdots & \ddots & \vdots \\
	d_G(x_n,x_1) & d_G(x_n,x_2) & \dots & 0
\end{bmatrix}
\]

که در آن \(d_G(x_i,x_j)\) فاصله ژئودزیک بین نقاط \(x_i\) و \(x_j\) است.

\paragraph{۳. اعمال الگوریتم \lr{MDS}}
در مرحله نهایی، هدف پیدا کردن نمایش داده‌ها در بعد پایین است به طوری که فواصل ژئودزیک حفظ شوند. برای این کار از \lr{Multidimensional Scaling (MDS)} کلاسیک استفاده می‌شود.

ابتدا ماتریس فاصله ژئودزیک به یک ماتریس شباهت تبدیل می‌شود:

\[
B = -\frac{1}{2} H D_G^2 H
\]

که در آن:

\[
H = I - \frac{1}{n}\mathbf{1}\mathbf{1}^T
\]

ماتریس مرکزسازی \lr{(centering matrix)} است. در رابطۀ بالا منظور از 1 همان بردار n در 1 است که شامل اعداد یک است. نتیجۀ ضرب این بردار در ترانهادۀ آن، یک ماتریس n در n می شود که تمام درایه های آن برابر با عدد یک است.

سپس با تجزیۀ مقادیر ویژه:

\[
B = V \Lambda V^T
\]

مختصات داده‌ها در فضایی با بعد پایین \(d\) به صورت زیر به دست می‌آید:

\[
Y = V_d \Lambda_d^{1/2}
\]

در این رابطه \(V_d\) شامل \(d\) بردار ویژه اول و \(\Lambda_d\) شامل بزرگ‌ترین مقادیر ویژه است. ماتریس \(Y\) داده‌ها را در بعد پایین نشان می‌دهد.

\paragraph{مزایا}
\begin{itemize}
	\item توانایی کشف ساختارهای غیرخطی  
	\item حفظ هندسه سراسری  
	\item راه‌حل بهینه و بدون مینیمم محلی  
\end{itemize}

\paragraph{معایب}
\begin{itemize}
	\item هزینه محاسباتی بسیار بالا \((O(N^3))\)  
	\item حساسیت به نویز و ایجاد مسیرهای کوتاه اشتباه   
\end{itemize}

\begin{itemize}
	\item \textbf{حساسیت به نویز تصاویر:} داده‌های خام تصویری دارای نویز بسیار بالایی هستند. الگوریتم ISOMAP به شدت به نویز حساس است و وجود نویز باعث پدیده اتصالات کوتاه (\lr{Short-circuiting}) در گراف همسایگی می‌شود که محاسبات فواصل ژئودزیک را دچار خطای شدید می‌کند.
	\item \textbf{پیچیدگی منیفلد و کمبود داده:} تصاویر طبیعی CIFAR-10 دارای ساختار منیفلد بسیار پیچیده‌ای هستند. الگوریتم ISOMAP با تعداد داده کم ($2000$ نمونه آموزشی) قادر به بازسازی هندسه سراسری این منیفلد نیست. در مقابل، \lr{PCA} صرفاً به دنبال جهات با بیشترین واریانس است و در فضاهای نویزی با ویژگی‌های خام، رفتار باثبات‌تری از خود نشان می‌دهد.
\end{itemize}
\subsubsection{بهینه‌سازی مدل‌ها با استفاده از جستجوی تصادفی \lr{(Random Search)} و \lr{Pipeline}}

\paragraph{روند پیاده‌سازی و استفاده از Pipeline}
برای یافتن بهترین پارامترهای مدل، از مفهوم \lr{Pipeline} در کتابخانه \lr{Scikit-Learn} استفاده شد. پایپ‌لاین به ما اجازه می‌دهد تا مراحل کاهش ابعاد (\lr{PCA} یا \lr{ISOMAP}) و طبقه‌بندی (KNN) را به صورت یک جریان واحد و پیوسته تعریف کنیم. این کار علاوه بر تمیزتر شدن کد، از نشت داده‌ها (\lr{Data Leakage}) در فرآیند ارزیابی متقاطع جلوگیری می‌کند. سپس با استفاده از \lr{RandomizedSearchCV}، فضای جستجویی از پارامترهای مختلف تعریف شد تا الگوریتم به صورت تصادفی ترکیبات مختلف را طی ارزیابی متقاطع ۳-لایه (\lr{3-Fold CV}) بررسی کرده و بهترین پیکربندی را بیابد.

\paragraph{تحلیل بهترین پارامترهای یافت شده برای PCA}
بهترین پارامترها در جستجوی تصادفی برای مدل PCA به این صورت انتخاب شد: تعداد مؤلفه‌ها برابر با $50$، تعداد همسایه‌ها $K=15$، وزن‌دهی \lr{uniform} و معیار فاصله \lr{euclidean}. 
\begin{itemize}
	\item \textbf{معیار فاصله اقلیدسی نسبت به منهتن:} انتخاب فاصله اقلیدسی ($L_2$) نسبت به منهتن ($L_1$) نشان‌دهنده آن است که در فضای کاهش‌یافته \lr{PCA}، فاصله مستقیم و هندسی بین مؤلفه‌ها اهمیت بیشتری در تفکیک کلاس‌ها دارد. بردارهای ویژگی متراکم، فاصله اقلیدسی معمولاً به دلیل حساسیت کمتر به چرخش محورها، رفتار پایدارتری در یافتن همسایه‌های واقعی از خود نشان می‌دهد.
	\item \textbf{وزن‌دهی \lr{uniform}:} این پارامتر نشان می‌دهد که در فضای خطی \lr{PCA}، رأی تمامی $15$ همسایه نزدیک ارزش یکسانی دارد و نیازی به جریمه کردن همسایه‌های دورتر نبوده است.
\end{itemize}

\paragraph{تحلیل بهترین پارامترهای یافت شده برای ISOMAP}
برای مدل \lr{ISOMAP}، جستجو به این پارامترها رسید: تعداد مؤلفه‌ها $50$، تعداد همسایه‌های گراف ISOMAP برابر با $10$، تعداد همسایه‌های KNN برابر با $10$ ($K=10$) و وزن‌دهی \lr{distance}.
\begin{itemize}
	\item \textbf{وزن‌دهی \lr{distance}:} برعکس \lr{PCA}، در اینجا مدل به وزن‌دهی مبتنی بر فاصله روی آورده است. دلیل این امر طبیعت غیرخطی خمینه‌های \lr{ISOMAP} است. در فضاهای غیرخطی و پیچیده، همسایه‌هایی که از نظر هندسی به \textbf{نمونه آزمون (تصویر ورودی)} نزدیک‌ترند، قابلیت اعتماد بسیار بالاتری نسبت به همسایه‌های دورتر در همان گروه \lr{$K$}-تایی دارند؛ لذا دادن وزن بیشتر به آن‌ها (رابطه معکوس با فاصله) دقت را بهبود بخشیده است.
\end{itemize}

\subsection{رمزگذار-رمزگشا \lr{(Autoencoder)}}
\subsubsection{استخراج ویژگی با استفاده از شبکه‌های خودرمزگذار (\lr{Autoencoders})}

برای بهبود استخراج ویژگی و غلبه بر محدودیت‌های روش‌های خطی مانند \lr{PCA}، از شبکه‌های عصبی عمیق در قالب معماری‌های خودرمزگذار (\lr{Autoencoder}) استفاده شد. این شبکه‌ها به صورت بدون ناظر (\lr{Unsupervised}) و با هدف کمینه‌سازی خطای بازسازی تصویر (تابع زیان \lr{MSE}) با بهینه‌ساز \lr{Adam} آموزش دیدند. در این رویکرد، داده‌ها به یک فضای پنهان (\lr{Latent Space}) با ابعاد کاهش‌یافته (۶۴ و ۱۰۰ بُعد) نگاشت شده و سپس ویژگی‌های استخراج‌شده برای طبقه‌بندی به الگوریتم \lr{KNN} داده شدند.

\paragraph{خودرمزگذار متراکم (\lr{Dense Autoencoder}) و از دست رفتن فواصل مکانی}

در معماری متراکم، تصاویر سه‌کاناله با اندازه $32\times32\times3$ ابتدا به یک بردار مسطح با طول \lr{3072} تبدیل می‌شوند. سپس این بردار از چند لایه کاملاً متصل (\lr{Fully Connected}) عبور داده می‌شود تا به یک نمایش فشرده در فضای پنهان برسد. در این تمرین، ساختار رمزگذار شامل سه لایه خطی با اندازه‌های \lr{512}، \lr{256} و در نهایت بعد فضای پنهان (\lr{latent\_dim}) است. در سمت رمزگشا نیز ساختار مشابهی به صورت معکوس برای بازسازی تصویر استفاده شده است. فعال‌ساز \lr{ReLU} در لایه‌های میانی و تابع \lr{Sigmoid} در خروجی استفاده شده است تا مقادیر پیکسل‌ها در بازه مناسب باقی بمانند.

ساختار پیاده‌سازی این مدل در چارچوب \lr{PyTorch} در کد زیر نشان داده شده است:

\begin{latin}
	\begin{lstlisting}[language=Python]
		class DenseAutoencoder(nn.Module):
		def __init__(self, latent_dim):
		super(DenseAutoencoder, self).__init__()
		self.encoder = nn.Sequential(
		nn.Linear(32 * 32 * 3, 512),
		nn.ReLU(),
		nn.Linear(512, 256),
		nn.ReLU(),
		nn.Linear(256, latent_dim)
		)
		self.decoder = nn.Sequential(
		nn.Linear(latent_dim, 256),
		nn.ReLU(),
		nn.Linear(256, 512),
		nn.ReLU(),
		nn.Linear(512, 32 * 32 * 3),
		nn.Sigmoid()
		)
		
		def encode(self, x):
		if len(x.shape) > 2:
		x = x.view(x.size(0), -1)
		return self.encoder(x)
		
		def forward(self, x):
		z = self.encode(x)
		out = self.decoder(z)
		return out
	\end{lstlisting}
\end{latin}

به دلیل تبدیل تصویر به بردار مسطح، ساختار مکانی پیکسل‌ها در این مدل از بین می‌رود. با این حال، وجود توابع فعال‌سازی غیرخطی باعث می‌شود مدل بتواند الگوهای پیچیده‌تری نسبت به روش‌های خطی مانند \lr{PCA} استخراج کند. نتایج تجربی نشان دادند که این مدل با فضای پنهان ۱۰۰ بعدی توانست به دقت حدود $44\%$ برسد که نسبت به \lr{PCA} عملکرد بهتری نشان می‌دهد.


\paragraph{خودرمزگذار کانولوشنی (\lr{CNN Autoencoder}) و کیفیت بازسازی}
برای حفظ ساختار مکانی تصاویر، یک معماری مبتنی بر شبکه‌های پیچشی (\lr{CNN}) شامل لایه‌های \lr{Conv2d} در رمزگذار و \lr{ConvTranspose2d} در رمزگشا طراحی شد. این شبکه ویژگی‌های محلی، لبه‌ها و بافت‌ها را به خوبی درک می‌کند؛ در نتیجه، نمودار تابع زیان (\lr{Loss}) آن سریع‌تر کاهش یافته و تصاویر بازسازی‌شده از کیفیت بصری بسیار بالاتری نسبت به مدل متراکم برخوردارند.

\paragraph{چرا عملکرد \lr{KNN} روی ویژگی‌های \lr{CNN} ضعیف‌تر شد؟}
با وجود بازسازی بهتر تصاویر در مدل \lr{CNN}، دقت طبقه‌بندی آن با \lr{KNN} در بهترین حالت به $40\%$ (در فضای ۶۴ بعدی) کاهش یافت. دلایل این افت عملکرد عبارتند از:
\begin{itemize}
	\item \textbf{ماهیت بدون ناظر بازسازی:} شبکه \lr{CNN} تنها برای بازسازی دقیق تصویر آموزش دیده است، نه برای تفکیک کلاس‌ها. بنابراین، ویژگی‌های استخراج‌شده شامل جزئیاتی مانند رنگ پس‌زمینه یا نورپردازی هستند که برای بازسازی مهم‌اند، اما کلاس تصویر را مشخص نمی‌کنند.
	\item \textbf{ضعف \lr{KNN} در فضاهای پیچیده:} الگوریتم \lr{KNN} بر اساس فاصله اقلیدسی کار می‌کند. فضای ویژگی‌های استخراج‌شده توسط \lr{CNN} بسیار انتزاعی است. ممکن است یک قورباغه روی چمن و یک گربه روی چمن از نظر ساختار تصویر به هم شبیه باشند و در فضای پنهانِ \lr{CNN} فاصله اقلیدسی کمی داشته باشند، که این امر \lr{KNN} را به اشتباه می‌اندازد. در مدل متراکم، به دلیل ترکیب کلی پیکسل‌ها، فاصله اقلیدسی کمی بهتر عمل کرده است.
\end{itemize}
اگر به جای بازسازی بدون ناظر، از یادگیری با ناظر (دادن برچسب‌ها به مدل) یا خودرمزگذارهای شرطی (\lr{Conditional Autoencoders}) استفاده می‌شد، فضای پنهان مجبور به فاصله‌گذاری معنایی میان کلاس‌ها می‌شد و دقت تفکیک بسیار بالاتر می‌رفت.
پیاده‌سازی این معماری به صورت زیر است:

\begin{latin}
	\begin{lstlisting}[language=Python]
		class ConvAutoencoder(nn.Module):
		
		def __init__(self, latent_dim):
		super(ConvAutoencoder, self).__init__()
		
		self.encoder_cnn = nn.Sequential(
		nn.Conv2d(3, 32, kernel_size=3, stride=2, padding=1),
		nn.BatchNorm2d(32),
		nn.ReLU(),
		
		nn.Conv2d(32, 64, kernel_size=3, stride=2, padding=1),
		nn.BatchNorm2d(64),
		nn.ReLU(),
		
		nn.Conv2d(64, 128, kernel_size=3, stride=2, padding=1),
		nn.BatchNorm2d(128),
		nn.ReLU(),
		)
		
		self.encoder_fc = nn.Linear(128 * 4 * 4, latent_dim)
		
		self.decoder_fc = nn.Linear(latent_dim, 128 * 4 * 4)
		
		self.decoder_cnn = nn.Sequential(
		nn.ConvTranspose2d(128, 64, kernel_size=4, stride=2, padding=1),
		nn.BatchNorm2d(64),
		nn.ReLU(),
		
		nn.ConvTranspose2d(64, 32, kernel_size=4, stride=2, padding=1),
		nn.BatchNorm2d(32),
		nn.ReLU(),
		
		nn.ConvTranspose2d(32, 3, kernel_size=4, stride=2, padding=1),
		nn.Sigmoid()
		)
		
		def encode(self, x):
		x = self.encoder_cnn(x)
		x = x.view(x.size(0), -1)
		z = self.encoder_fc(x)
		return z
		
		def forward(self, x):
		x = self.encoder_cnn(x)
		x = x.view(x.size(0), -1)
		z = self.encoder_fc(x)
		
		z = self.decoder_fc(z)
		z = z.view(z.size(0), 128, 4, 4)
		out = self.decoder_cnn(z)
		return out
	\end{lstlisting}
\end{latin}

\subsection{خود رمزگذار متغیر \lr{(Variational Auto Encoder)}}
\subsubsection{مروری بر خودرمزگذارهای متغیر (\lr{Variational Autoencoders - VAEs})}

خودرمزگذارهای متغیر (\lr{VAE}) که برای اولین بار توسط \lr{Kingma} و \lr{Welling} در سال ۲۰۱۳ معرفی شدند، دسته‌ای از مدل‌های زایای عمیق (\lr{Deep Generative Models}) هستند. برخلاف خودرمزگذارهای سنتی که داده‌ها را به یک بردار ثابت و قطعی در فضای پنهان نگاشت می‌کنند، \lr{VAE}ها داده‌ها را به یک \textbf{توزیع احتمال} نگاشت می‌کنند. این ویژگی باعث می‌شود فضای پنهان پیوسته و منظم باشد و بتوان از آن برای تولید داده‌های جدید و شبیه به داده‌های آموزشی استفاده کرد.

\paragraph{معماری و شیوه عملکرد}
یک \lr{VAE} از دو بخش اصلی تشکیل شده است:
\begin{itemize}
	\item \textbf{رمزگذار (\lr{Encoder}):} داده ورودی $x$ را دریافت کرده و به جای یک بردار ویژگی ثابت، پارامترهای یک توزیع احتمال (معمولاً میانگین $\mu$ و واریانس $\sigma^2$ از یک توزیع نرمال) را برای فضای پنهان تخمین می‌زند.
	\item \textbf{رمزگشا (\lr{Decoder}):} یک نمونه تصادفی $z$ را از توزیع به دست آمده در فضای پنهان می‌گیرد و سعی می‌کند داده اولیه را بازسازی کند (تولید $\hat{x}$).
\end{itemize}

\paragraph{ترفند بازپارامترسازی (\lr{Reparameterization Trick})}
برای اینکه شبکه عصبی بتواند از طریق الگوریتم پس‌انتشار خطا (\lr{Backpropagation}) آموزش ببیند، باید بتوان از عملیات نمونه‌برداری مشتق گرفت. از آنجایی که نمونه‌برداری یک فرآیند تصادفی است و مشتق‌پذیر نیست، از ترفند بازپارامترسازی استفاده می‌شود. در این روش، متغیر پنهان $z$ به صورت زیر تعریف می‌شود:
$$ z = \mu + \sigma \odot \epsilon $$
که در آن $\epsilon \sim \mathcal{N}(0, I)$ یک نویز تصادفی از توزیع نرمال استاندارد است و $\odot$ ضرب درایه به درایه را نشان می‌دهد. با این کار، تصادفی بودن به متغیر $\epsilon$ منتقل شده و گرادیان‌ها می‌توانند از $\mu$ و $\sigma$ عبور کنند.

\paragraph{تابع زیان (\lr{Loss Function})}
تابع زیان در \lr{VAE}ها از دو بخش اساسی تشکیل شده است:
$$ \mathcal{L} = \text{Reconstruction Loss} + D_{KL}(q_\phi(z|x) || p(z)) $$
\begin{enumerate}
	\item \textbf{خطای بازسازی (\lr{Reconstruction Loss}):} میزان شباهت داده تولید شده $\hat{x}$ به داده ورودی $x$ را می‌سنجد (معمولاً با استفاده از \lr{MSE} یا \lr{Cross-Entropy}).
	\item \textbf{واگرایی کولبک-لایبلر (\lr{KL Divergence}):} این جمله به عنوان یک تنظیم‌کننده (\lr{Regularizer}) عمل می‌کند و توزیع تخمین زده شده توسط رمزگذار $q_\phi(z|x)$ را مجبور می‌کند تا حد امکان به توزیع پیشین (\lr{Prior}) که معمولاً یک توزیع نرمال استاندارد $p(z) = \mathcal{N}(0, I)$ است، نزدیک شود.
\end{enumerate}
ترکیب این دو خطا باعث می‌شود تا فضای پنهان هم برای بازسازی داده‌ها مناسب باشد (حفظ اطلاعات) و هم ساختار پیوسته‌ای داشته باشد تا با درون‌یابی (\lr{Interpolation}) بین نقاط آن، داده‌های معناداری تولید شود.

\subsubsection{پیاده‌سازی معماری‌های \lr{VAE}}

برای بررسی عملکرد فضای پنهان پیوسته، دو مدل \lr{DenseVAE} و \lr{CNN\_VAE} پیاده‌سازی شدند. 

\paragraph{معماری \lr{DenseVAE}}
در این مدل، تصاویر به بردارهای یک‌بعدی تبدیل شده و از لایه‌های تمام‌متصل برای رمزگذاری و رمزگشایی استفاده می‌شود. خروجی رمزگذار شامل دو بردار $\mu$ و $\log(\sigma^2)$ است که با ترفند بازپارامترسازی به $z$ تبدیل می‌شوند.

\begin{latin}
	\begin{lstlisting}[language=Python, basicstyle=\ttfamily\scriptsize]
		class DenseVAE(nn.Module):
		def __init__(self, z_size):
		super(DenseVAE, self).__init__()
		self.z_size = z_size
		
		self.encoder = nn.Sequential(
		nn.Linear(32 * 32 * 3, 512),
		nn.ReLU(),
		nn.Linear(512, 256),
		nn.ReLU()
		)
		self.fc_mu = nn.Linear(256, z_size)
		self.fc_logvar = nn.Linear(256, z_size)
		
		self.decoder = nn.Sequential(
		nn.Linear(z_size, 256),
		nn.ReLU(),
		nn.Linear(256, 512),
		nn.ReLU(),
		nn.Linear(512, 32 * 32 * 3),
		nn.Sigmoid()
		)
		
		def encode(self, x):
		
		if len(x.shape) > 2:
		x = x.view(x.size(0), -1)
		h = self.encoder(x)
		return self.fc_mu(h), self.fc_logvar(h)
		
		def reparameterize(self, mu, logvar):
		std = torch.exp(0.5 * logvar)
		eps = torch.randn_like(std)
		return mu + eps * std
		
		def forward(self, x):
		if len(x.shape) > 2:
		x = x.view(x.size(0), -1)
		mu, logvar = self.encode(x)
		z = self.reparameterize(mu, logvar)
		out = self.decoder(z)
		return out, mu, logvar
	\end{lstlisting}
\end{latin}

\paragraph{معماری \lr{CNN\_VAE}}
این مدل با استفاده از لایه‌های کانولوشن (\lr{Conv2d}) در رمزگذار و کانولوشن ترانهاده (\lr{ConvTranspose2d}) در رمزگشا، ساختار مکانی تصاویر را حفظ می‌کند.

\begin{latin}
	\begin{lstlisting}[language=Python, basicstyle=\ttfamily\scriptsize]
		class VAE(nn.Module):
		def __init__(self, z_size, image_size=32, channel_num=3, kernel_num=128):
		super().__init__()
		self.image_size = image_size
		self.channel_num = channel_num
		self.kernel_num = kernel_num
		self.z_size = z_size
		
		
		self.encoder = nn.Sequential(
		self._conv(channel_num, kernel_num // 4),
		self._conv(kernel_num // 4, kernel_num // 2),
		self._conv(kernel_num // 2, kernel_num),
		)
		
		self.feature_size = image_size // 8
		self.feature_volume = kernel_num * (self.feature_size ** 2)
		
		self.q_mean = nn.Linear(self.feature_volume, z_size)
		self.q_logvar = nn.Linear(self.feature_volume, z_size)
		self.project = nn.Linear(z_size, self.feature_volume)
		
		
		self.decoder = nn.Sequential(
		self._deconv(kernel_num, kernel_num // 2),
		self._deconv(kernel_num // 2, kernel_num // 4),
		self._deconv(kernel_num // 4, channel_num),
		nn.Sigmoid()
		)
		
		def forward(self, x):
		mean, logvar, z = self.encode(x)
		z_projected = self.project(z).view(
		-1, self.kernel_num,
		self.feature_size,
		self.feature_size,
		)
		x_reconstructed = self.decoder(z_projected)
		return (mean, logvar), x_reconstructed
		
		def encode(self, x):
		encoded = self.encoder(x)
		mean, logvar = self.q(encoded)
		z = self.z(mean, logvar)
		return mean, logvar, z
		
		def q(self, encoded):
		unrolled = encoded.view(-1, self.feature_volume)
		return self.q_mean(unrolled), self.q_logvar(unrolled)
		
		def z(self, mean, logvar):
		std = torch.exp(0.5 * logvar)
		eps = torch.randn_like(std)
		return mean + eps * std
		
		def _conv(self, channel_size, kernel_num):
		return nn.Sequential(
		nn.Conv2d(channel_size, kernel_num, kernel_size=4, stride=2, padding=1),
		nn.BatchNorm2d(kernel_num),
		nn.LeakyReLU(0.2),
		)
		
		def _deconv(self, channel_num, kernel_num):
		return nn.Sequential(
		nn.ConvTranspose2d(channel_num, kernel_num, kernel_size=4, stride=2, padding=1),
		nn.BatchNorm2d(kernel_num),
		nn.LeakyReLU(0.2),
		)
	\end{lstlisting}
\end{latin}

\subsubsection{تحلیل و مقایسه نتایج}

بررسی نتایج به دست آمده حاوی نکات مهمی درباره ماهیت \lr{VAE}ها و تناسب آن‌ها با طبقه‌بند \lr{KNN} است.

\paragraph{۱. مقایسه \lr{Dense VAE} با \lr{Dense AE}}
در نتایج قبلی، خودرمزگذار \lr{Dense} معمولی به دقت حدود $0.44$ رسیده بود، اما مدل \lr{Dense VAE} در بهترین حالت (۶۴ بعدی) به دقت $0.42$ و در ۱۰۰ بعدی به $0.41$ رسیده است. این افت اندک به دلیل وجود جمله \textbf{واگرایی کولبک–لایبلر (\lr{KL Divergence})} در تابع زیان \lr{VAE} است که به عنوان یک تنظیم‌کننده قوی عمل کرده و توزیع ویژگی‌ها را به توزیع نرمال استاندارد نزدیک می‌کند. در نتیجه مرز بین کلاس‌ها تا حدی کاهش یافته و عملکرد \lr{KNN} که مبتنی بر فاصله اقلیدسی است کاهش می‌یابد.

\paragraph{۲. مقایسه \lr{CNN VAE} و \lr{Dense VAE}}
مدل \lr{CNN VAE} دقت‌های پایین‌تری ($0.38$ برای ۶۴ بعدی و $0.36$ برای ۱۰۰ بعدی) ثبت کرده است. اگرچه این مدل‌ها در بازسازی تصاویر عملکرد بسیار بهتری دارند، اما فضای ویژگی ایجاد شده توسط \lr{CNN} بسیار انتزاعی و غیرخطی است و فاصله اقلیدسی مورد استفاده در \lr{KNN} نمی‌تواند ساختار معنایی این فضا را به خوبی مدل کند.

\paragraph{۳. تأثیر ابعاد فضای پنهان}
در هر دو مدل، افزایش بعد فضای پنهان از ۶۴ به ۱۰۰ بهبودی ایجاد نکرده و حتی کمی باعث افت دقت شده است. در \lr{VAE}ها بخشی از ابعاد اضافی ممکن است صرف مدل‌سازی نویز توزیع پیشین شود و اطلاعات تمایزدهنده کلاس‌ها را افزایش ندهد.

در مجموع، هرچند \lr{VAE}ها در یادگیری توزیع داده‌ها و بازسازی تصاویر بسیار قدرتمند هستند اما در تنظیمات بدون نظارت لزوماً فضای ویژگی مناسبی برای طبقه‌بندی مبتنی بر فاصله مانند \lr{KNN} تولید نمی‌کنند. بنابراین استفاده از مدل‌های شرطی مانند \lr{Conditional VAE} یا به‌کارگیری طبقه‌بندهای قوی‌تر مانند \lr{SVM} یا شبکه‌های کاملاً متصل می‌تواند عملکرد بهتری ایجاد کند.

مدل‌های مبتنی بر کانولوشن (\lr{CNN\_VAE}) در هر دو بعد ۶۴ و ۱۰۰، در نهایت به مقدار زیان کمتری نسبت به مدل‌های \lr{Dense} رسیده‌اند. این موضوع تأیید می‌کند که لایه‌های کانولوشن به دلیل حفظ روابط مکانی در تصاویر \lr{CIFAR-10}، توانایی بسیار بالاتری در بازسازی دقیق تصویر و استخراج ویژگی‌های کلیدی دارند.

\paragraph{تأثیر ابعاد فضای پنهان}

در مدل‌های \lr{Dense}، افزایش بعد از ۶۴ به ۱۰۰ منجر به افزایش جزئی در مقدار زیان شده است (نمودار قرمز بالاتر از سبز)، که می‌تواند نشان‌دهنده پیچیدگی بیشتر در بهینه‌سازی فضای ۱۰۰ بعدی در ساختار تمام‌متصل باشد. در مقابل، در مدل‌های \lr{CNN}، هر دو بعد ۶۴ و ۱۰۰ رفتار بسیار نزدیکی به هم دارند و با اختلافی ناچیز در انتهای اپوک دهم همگرا شده‌اند، که نشان‌دهنده پایداری بیشتر معماری کانولوشنی نسبت به تغییر ابعاد است.

\subsubsection{تحلیل نمودار پراکندگی فضای پنهان مدل \lr{ConvVAE} (فضای ۱۰۰ بعدی)}

این نمودار نشان‌دهنده‌ی توزیع داده‌های آموزشی \lr{CIFAR-10} در فضای پنهان (\lr{Latent Space}) مدل \lr{Convolutional VAE} است. بر اساس کد ارائه شده، مختصات رسم شده تنها مربوط به دو بُعد اول بردار میانگین (یعنی $\mu_1$ و $\mu_2$) از یک فضای پنهان ۱۰۰ بعدی است. در ادامه رفتار این فضا از زوایای مختلف تحلیل شده است:

\paragraph{۱. تأثیر واگرایی کولبک-لایبلر (\lr{KL Divergence}) و مرکزیت داده‌ها}

مهم‌ترین مشاهدۀ بصری در این نمودار، تجمع کُروی و متراکم تمامی نقاط حول مبدأ مختصات $(0,0)$ است. این رفتار مستقیماً ناشی از ترم واگرایی \lr{KL} در تابع زیان (\lr{Loss Function}) مدل \lr{VAE} است. این ترم، توزیع فضای پنهان $q(z|x)$ را مجبور می‌کند تا حد امکان به توزیع پیشین (\lr{Prior}) که یک توزیع نرمال استاندارد $\mathcal{N}(0,1)$ است، نزدیک شود. به همین دلیل، مقادیر میانگین ($\mu$) استخراج‌شده برای تصاویر مختلف، همگی در محدوده‌ای نزدیک به صفر (بین $-4$ تا $4$ برای $\mu_1$ و $-2$ تا $2$ برای $\mu_2$) فشرده شده‌اند و فضا پیوستگی بالایی دارد.

\paragraph{۲. هم‌پوشانی شدید کلاس‌ها (فقدان جدایی‌پذیری خطی)}

همان‌طور که از رنگ‌بندی نقاط (بر اساس ۱۰ کلاس مختلف \lr{CIFAR-10}) مشخص است، هیچ‌گونه مرزبندی واضح یا خوشه‌بندی (\lr{Clustering}) تفکیک‌پذیری بین کلاس‌ها دیده نمی‌شود و رنگ‌ها کاملاً در هم آمیخته‌اند. دلایل این پدیده عبارتند از:

\begin{itemize}
	\item \textbf{ماهیت بدون ناظر (\lr{Unsupervised}) بودن \lr{VAE}:} شبکه‌ی \lr{VAE} بدون اطلاع از برچسب کلاس‌ها آموزش می‌بیند و تنها به دنبال بازسازی پیکسل‌های تصویر است. بنابراین، ممکن است دو تصویر از کلاس‌های متفاوت (مثلاً «سگ» و «اسب») که پس‌زمینه‌ی سبز مشابهی دارند، در فضای پنهان در کنار یکدیگر قرار گیرند.
	\item \textbf{فقدان حاشیه (\lr{Margin}):} برخلاف روش‌های نظارت‌شده (مثل \lr{Contrastive Learning} یا شبکه‌های دارای \lr{Cross-Entropy Loss}) که سعی در دور کردن کلاس‌های متفاوت دارند، \lr{VAE} برای تولید یک فضای پیوسته (جهت تولید تصاویر جدید) تلاش می‌کند، نه یک فضای تفکیک‌شده.
\end{itemize}

\paragraph{۳. محدودیت تصویرسازی دو بُعد از صد بُعد (\lr{Dimensionality Projection})}

کد رسم نمودار (\lr{\texttt{z\_train\_features[:, 0]}} و \lr{\texttt{z\_train\_features[:, 1]}}) نشان می‌دهد که ما تنها در حال مشاهده‌ی ۲ بُعد اول از ۱۰۰ بُعد استخراج‌شده هستیم. این یک برش بسیار ساده‌انگارانه از فضای داده‌هاست. در حالی که این دو بُعد به شدت در هم تنیده به نظر می‌رسند، ممکن است در ۹۸ بُعد دیگر، الگوهای تمایزدهنده‌ای وجود داشته باشد. برای مشاهده بهتر خوشه‌ها در فضاهای با ابعاد بالا، استفاده از الگوریتم‌های کاهش ابعاد مانند \lr{t-SNE} یا \lr{UMAP} روی کل بردار ۱۰۰ بعدی، بسیار منطقی‌تر از انتخاب صرفاً دو ویژگی اول است.

\paragraph{۴. ارتباط تصویر با افت دقت طبقه‌بند \lr{KNN}}

این نمودار دلیل اصلی افت عملکرد الگوریتم \lr{KNN} (با دقت $0.38$) روی ویژگی‌های \lr{ConvVAE} را به وضوح توضیح می‌دهد. الگوریتم \lr{KNN} بر پایه فاصله اقلیدسی عمل می‌کند. وقتی نمونه‌های کلاس‌های مختلف تا این حد در فضای پنهان فشرده و در هم تنیده شده‌اند، یافتن همسایه‌های نزدیک که متعلق به همان کلاس هدف باشند، به شدت دشوار می‌شود. این فشردگی که برای تولید تصاویر (\lr{Generation}) یک مزیت است، برای وظیفه‌ی طبقه‌بندی (\lr{Classification}) یک چالش بزرگ محسوب می‌شود.

تصویر و کد ارائه شده، نمایشی از چگونگی عملکرد شبکه رمزگشا (\lr{Decoder}) مدل \lr{ConvVAE} در تبدیل بردارهای انتزاعی فضای پنهان به تصاویر معنادار است. این خروجی نشان‌دهنده‌ی روند تولید تصویر و تأثیر تغییرات مختصات فضای پنهان بر ویژگی‌های بصری خروجی است. در ادامه جزئیات این فرآیند تحلیل شده است:

\paragraph{۱. منطق کد و نحوه‌ی پیمایش فضا}
در کد نوشته شده، یک شبکه (\lr{Grid}) دو بعدی $10 \times 10$ با استفاده از تابع \lr{\texttt{np.linspace}} بین مقادیر $-2.0$ تا $2.0$ ایجاد شده است. با توجه به اینکه فضای پنهان این مدل ۱۰۰ بعدی است، کد ابتدا یک بردار تماماً صفر ایجاد می‌کند ($z_i = 0$ برای $i > 2$). سپس تنها دو بُعد اول این بردار ($z_1$ و $z_2$) را با مقادیر این شبکه دو بعدی مقداردهی می‌کند. این بردار به عنوان ورودی به \lr{Decoder} داده می‌شود تا تصویر متناظر با آن نقطه از فضا تولید شود.

\paragraph{۲. پیوستگی و تغییرات هموار (\lr{Smooth Interpolation})}
مهم‌ترین ویژگی مشاهده شده در تصویر خروجی، تغییرات بسیار نرم و پیوسته رنگ‌ها و ساختارهای کلی (مانند تغییر از تونالیته تیره به روشن یا تغییر رنگ زمینه) با حرکت در طول سطرها و ستون‌ها است. این رفتار نشان‌دهنده‌ی موفقیت مدل \lr{VAE} در ایجاد یک فضای پنهان پیوسته است. برخلاف \lr{Autoencoder}های معمولی که فضای پنهان گسسته‌ای دارند، استفاده از توزیع نرمال و ترم \lr{KL Divergence} در \lr{VAE} تضمین می‌کند که نقاط نزدیک به هم در فضای پنهان، تصاویر شبیه به هم تولید کنند.

\paragraph{۳. دلیل محو بودن و عدم وضوح تصاویر تولید شده}
تصاویر موجود در این منیفولد بسیار تار (\lr{Blurry}) هستند و شکل اشیاء مشخصی از کلاس‌های \lr{CIFAR-10} (مانند ماشین، سگ و...) در آن‌ها دیده نمی‌شود. این مسئله دو دلیل عمده دارد:
\begin{itemize}
	\item \textbf{صفر بودن ۹۸ بُعد دیگر:} در این کد، برای رسم یک تصویر دو بعدی، ۹۸ ویژگی دیگر فضای پنهان روی عدد صفر تنظیم شده‌اند. در یک مدل ۱۰۰ بعدی، اطلاعات حیاتی مربوط به لبه‌ها، بافت‌ها و شکل دقیق اشیاء در تمام ۱۰۰ بُعد پخش شده است. با صفر کردن آن‌ها، ما تنها در حال مشاهده یک «برش بسیار نازک» از این فضای پیچیده هستیم که طبیعتاً فاقد جزئیات کافی برای بازسازی یک تصویر کامل است.
	\item \textbf{ذات مدل‌های \lr{VAE}:} مدل‌های \lr{VAE} به دلیل استفاده از خطای بازسازی میانگین مربعات (\lr{MSE})، به طور ذاتی تمایل به تولید تصاویر محوتر نسبت به مدل‌هایی مانند \lr{GAN} دارند.
\end{itemize}

\subsection{مدل های انتشار \lr{(Diffusion Models)}}
\subsubsection{مدل‌های انتشار (\lr{Diffusion Models}) و مقایسه آن‌ها با \lr{VAE}}

در سال‌های اخیر، کلاس جدیدی از شبکه‌های مولد به نام \textbf{مدل‌های انتشار} (\lr{Diffusion Models}) معرفی شده‌اند که در زمینه تولید تصاویر با کیفیت بالا، به استانداردی جدید (\lr{State-of-the-Art}) تبدیل شده‌اند. این مدل‌ها توانسته‌اند بر بسیاری از محدودیت‌های مدل‌های پیشین مانند \lr{GAN}ها (ناپایداری در آموزش) و \lr{VAE}ها (تولید تصاویر محو) غلبه کنند.

\paragraph{نحوه عملکرد مدل‌های انتشار (\lr{Diffusion Models})}
ایده‌ی اصلی مدل‌های انتشار، الهام گرفته از فیزیک ترمودینامیک غیرتعادلی است. این مدل‌ها فرآیند تولید داده را به صورت یک فرآیند مارکوفی دو مرحله‌ای (رفت و برگشت) مدل‌سازی می‌کنند:

\begin{enumerate}
	\item \textbf{فرآیند رفت (\lr{Forward Process}):} در این مرحله، به تدریج و طی $T$ گام زمانی، نویز گاوسی به تصویر اصلی ($x_0$) اضافه می‌شود تا در نهایت به یک نویز خالص ($x_T$) تبدیل گردد. توزیع احتمال انتقال به صورت زیر تعریف می‌شود:
	\[ q(x_t | x_{t-1}) = \mathcal{N}(x_t; \sqrt{1 - \beta_t} x_{t-1}, \beta_t \mathbf{I}) \]
	که در آن $\beta_t$ واریانس نویز در گام $t$ است.
	
	\item \textbf{فرآیند برگشت (\lr{Reverse Process}):} شبکه عصبی (معمولاً یک معماری \lr{U-Net}) آموزش می‌بیند تا فرآیند رفت را معکوس کند؛ یعنی با دریافت یک نویز خالص ($x_T$)، به صورت گام‌به‌گام نویز را حذف کرده و تصویر اصلی را بازسازی کند:
	\[ p_\theta(x_{t-1} | x_t) = \mathcal{N}(x_{t-1}; \mu_\theta(x_t, t), \Sigma_\theta(x_t, t)) \]
	شبکه با کمینه‌سازی خطای میانگین مربعات (\lr{MSE}) بین نویز واقعی اضافه شده و نویز پیش‌بینی شده، آموزش می‌بیند.
\end{enumerate}

\paragraph{مدل‌های انتشار پنهان (\lr{Latent Diffusion Models})}
برای رفع مشکل کندی فرآیند تولید در مدل‌های انتشار استاندارد، معماری جدیدی به نام \lr{LDM} معرفی شد (که پایه و اساس \lr{Stable Diffusion} است). در این رویکرد، به جای اعمال فرآیند انتشار روی فضای پیکسلی، ابتدا یک \lr{Autoencoder} (مشابه \lr{VAE}) تصویر را به یک فضای پنهان فشرده ($z$) نگاشت می‌کند و فرآیند انتشار در این فضای پنهان با ابعاد کمتر انجام می‌شود.

\paragraph{مقایسه جامع \lr{VAE} و \lr{Diffusion Models}}
اگرچه هر دو مدل بر پایه تئوری احتمالات و استنتاج متغیر (\lr{Variational Inference}) بنا شده‌اند، تفاوت‌های بنیادینی دارند:

\begin{itemize}
	\item \textbf{کیفیت و وضوح تصاویر (\lr{Fidelity}):}
	مدل‌های \lr{VAE} به دلیل استفاده از تابع زیان مبتنی بر بازسازی (مانند \lr{MSE}) و فشرده‌سازی اطلاعات در یک گلوگاه (\lr{Bottleneck}) منفرد، معمولاً تصاویری محو (\lr{Blurry}) تولید می‌کنند. در مقابل، مدل‌های \lr{Diffusion} به دلیل فرآیند تدریجی کاهش نویز در صدها گام، قادر به تولید جزئیات بسیار دقیق (\lr{High-Fidelity}) هستند.
	
	\item \textbf{ساختار فضای پنهان (\lr{Latent Space}):}
	در \lr{VAE}، فضای پنهان ابعاد بسیار کمتری نسبت به تصویر ورودی دارد و برای کارهایی مانند پیمایش در فضا (\lr{Interpolation}) عالی است. اما در مدل‌های انتشار استاندارد، فضای متغیرهای پنهان دارای ابعادی برابر با تصویر اصلی است و فشرده‌سازی ابعادی (بدون لایه‌های میانی \lr{LDM}) رخ نمی‌دهد.
	
	\item \textbf{سرعت استنتاج و تولید (\lr{Sampling Speed}):}
	تولید تصویر در \lr{VAE} تنها با یک بار عبور از شبکه رمزگشا انجام می‌شود و بسیار سریع است. اما \lr{Diffusion} نیازمند عبور مکرر (از ده‌ها تا هزاران بار) از شبکه برای حذف نویز است که فرآیند تولید را بسیار کندتر می‌کند.
\end{itemize}
